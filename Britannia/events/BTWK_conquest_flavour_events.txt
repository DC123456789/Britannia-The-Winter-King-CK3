# Master controller event
conquest_flavour_anglosaxon.0001 = {
	scope = none
	content_source = dlc_BTWK
	hidden = yes
	
	immediate = {
		title:e_aengland.holder ?= { 
			save_scope_as = conqueror
			if = {
				limit = {
					title:e_aengland = { NOT = { has_variable = aengland_conquered } }
					BTWK_HK_aengland_unite_aengland_decision_control_requirement_trigger = yes
				}
				title:e_aengland = { set_variable = aengland_conquered }
				trigger_event = { id = conquest_flavour_anglosaxon.0002 }
				every_player = {
					limit = { NOT = { this = PREV } }
					trigger_event = { id = conquest_flavour_anglosaxon.0002 }
				}
			}
			if = {
				limit = {
					title:e_aengland = { NOT = { has_variable = wales_conquered } }
					completely_controls_region = btwk_custom_wales
					always = no		# Disabled for now
				}
				title:e_aengland = { set_variable = wales_conquered }
				trigger_event = { id = conquest_flavour_anglosaxon.0003 }
				every_player = {
					limit = { NOT = { this = PREV } }
					trigger_event = { id = conquest_flavour_anglosaxon.0003 }
				}
			}
			if = {
				limit = {
					title:e_aengland = { NOT = { has_variable = britannia_conquered } }
					completely_controls_region = btwk_world_britannia
				}
				title:e_aengland = { set_variable = britannia_conquered }
				trigger_event = { id = conquest_flavour_anglosaxon.0004 }
				every_player = {
					limit = { NOT = { this = PREV } }
					trigger_event = { id = conquest_flavour_anglosaxon.0004 }
				}
			}
		}
	}
}

# Master controller event
conquest_flavour_briton.0001 = {
	scope = none
	content_source = dlc_BTWK
	hidden = yes
	
	immediate = {
		title:c_londinium.holder.top_liege = {
			save_scope_as = conqueror
			if = {
				limit = {
					title:e_britannia = { NOT = { has_variable = anglo_saxons_defeated_south } }
					NOT = {
						any_county_in_region = {
							region = btwk_world_britannia_south
							holder.top_liege = {
								OR = {
									is_unromanized_germanic_culture_trigger = yes
									primary_title.empire = title:e_aengland
								}
							}
						}
					}
					# Need to wait until all relevant Anglo-Saxon invasions have been triggered and defeated
					OR = {
						NOT = { exists = struggle:britannia_struggle }
						NOR = {
							BTWK_rise_of_anglo_saxon_kingdom_is_kingdom_visible_trigger = {
								KINGDOM = title:k_lindsey
							}
							BTWK_rise_of_anglo_saxon_kingdom_is_kingdom_visible_trigger = {
								KINGDOM = title:k_mercia
							}
							BTWK_rise_of_anglo_saxon_kingdom_is_kingdom_visible_trigger = {
								KINGDOM = title:k_wessex
							}
						}
					}
				}
				title:e_britannia = { set_variable = anglo_saxons_defeated_south }
				# Done here to make sure that no weirdness happens if both events are triggered at once
				if = {
					limit = { title:e_britannia = { has_variable = anglo_saxons_defeated_north } }
					title:e_britannia = { set_variable = anglo_saxons_defeated_all }
				}
				trigger_event = { id = conquest_flavour_briton.0002 }
				every_player = {
					limit = { NOT = { this = PREV } }
					trigger_event = { id = conquest_flavour_briton.0002 }
				}
			}
		}
		title:c_eburacum.holder.top_liege = {
			if = {
				limit = {
					title:e_britannia = { NOT = { has_variable = anglo_saxons_defeated_north } }
					NOT = {
						any_county_in_region = {
							region = btwk_world_britannia_hen_ogledd
							holder.top_liege = {
								OR = {
									is_unromanized_germanic_culture_trigger = yes
									primary_title.empire = title:e_aengland
								}
							}
						}
					}
					# Need to wait until all relevant Anglo-Saxon invasions have been triggered and defeated
					OR = {
						NOT = { exists = struggle:britannia_struggle }
						NOR = {
							BTWK_rise_of_anglo_saxon_kingdom_is_kingdom_visible_trigger = {
								KINGDOM = title:k_deira
							}
							BTWK_rise_of_anglo_saxon_kingdom_is_kingdom_visible_trigger = {
								KINGDOM = title:k_bernicia
							}
						}
					}
				}
				title:e_britannia = { set_variable = anglo_saxons_defeated_north }
				if = {
					limit = { title:e_britannia = { has_variable = anglo_saxons_defeated_south } }
					title:e_britannia = { set_variable = anglo_saxons_defeated_all }
				}
				trigger_event = { id = conquest_flavour_briton.0003 }
				every_player = {
					limit = { NOT = { this = PREV } }
					trigger_event = { id = conquest_flavour_briton.0003 }
				}
			}
		}
		title:e_britannia.holder ?= {
			save_scope_as = conqueror
			if = {
				limit = {
					title:e_britannia = { NOT = { has_variable = britannia_conquered } }
					completely_controls_region = btwk_world_britannia
					always = no		# Disabled for now
				}
				title:e_britannia = { set_variable = britannia_conquered }
				trigger_event = { id = conquest_flavour_briton.0004 }
				every_player = {
					limit = { NOT = { this = PREV } }
					trigger_event = { id = conquest_flavour_briton.0004 }
				}
			}
		}
	}
}

# Anglo-Saxons defeated in the South
conquest_flavour_briton.0002 = {
	type = character_event
	content_source = dlc_BTWK
	window = big_event_window
	title = {
		first_valid = {
			triggered_desc = {
				trigger = { title:e_britannia = { has_variable = anglo_saxons_defeated_all } }
				desc = conquest_flavour_briton.0002.t.total_defeat
			}
			desc = conquest_flavour_briton.0002.t
		}
	}
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					exists = scope:southern_britannia_largest_controlled_culture
					title:e_britannia = { has_variable = anglo_saxons_defeated_all } 
				}
				desc = conquest_flavour_briton.0002.desc.total_defeat.non_briton_controlling_culture
			}
			triggered_desc = {
				trigger = { title:e_britannia = { has_variable = anglo_saxons_defeated_all } }
				desc = conquest_flavour_briton.0002.desc.total_defeat
			}
			triggered_desc = {
				trigger = { exists = scope:southern_britannia_largest_controlled_culture }
				desc = conquest_flavour_briton.0002.desc.non_briton_controlling_culture
			}
			desc = conquest_flavour_briton.0002.desc
		}
	}
	theme = war
	override_background = farmland
	
	left_portrait = {
		character = scope:conqueror
		animation = personality_war_over_win
	}
	
	immediate = {
		character:80 = {
			save_scope_as = aelle_of_anglia
		}
		# Figure out which culture controls most of southern Britannia
		if = {
			limit = {
				NOT = {
					any_county_in_region = {
						geographical_region = btwk_world_britannia_south
						percent >= 0.5
						holder.top_liege = { is_britannic_culture_trigger = yes }
					}
				}
			}
			every_culture_global = {
				set_variable = {
					name = southern_britannia_controlled_counties
					value = 0
				}
			}
			every_county_in_region = {
				region = btwk_world_britannia_south
				holder.top_liege.culture = {
					change_variable = {
						name = southern_britannia_controlled_counties
						add = 1
					}
				}
			}
			ordered_culture_global = {
				order_by = this.var:southern_britannia_controlled_counties
				save_scope_as = southern_britannia_largest_controlled_culture
			}
		}
	}

	option = {
		name = {
			trigger = {
				culture = {
					OR = {
						has_cultural_pillar = heritage_brythonic
						has_cultural_pillar = heritage_latin
					}
				}
				title:e_britannia = { has_variable = anglo_saxons_defeated_all }
			}
			text = conquest_flavour_briton.0002.a.briton_roman.total_defeat
		}
		name = {
			trigger = {
				culture = {
					OR = {
						has_cultural_pillar = heritage_brythonic
						has_cultural_pillar = heritage_latin
					}
				}
				NOT = { title:e_britannia = { has_variable = anglo_saxons_defeated_all } }
			}
			text = conquest_flavour_briton.0002.a.briton_roman
		}
		name = {
			trigger = {
				NOT = { this = scope:conqueror }
				culture = { 
					has_cultural_pillar = heritage_west_germanic
				}
				any_held_title = {
					tier = prev.primary_title.tier
					title_capital_county.title_province = { geographical_region = btwk_world_britannia }
				}
				title:e_britannia = { has_variable = anglo_saxons_defeated_all }
			}
			text = conquest_flavour_briton.0002.a.anglosaxon.total_defeat
		}
		name = {
			trigger = {
				NOT = { this = scope:conqueror }
				culture = { 
					has_cultural_pillar = heritage_west_germanic
				}
				any_held_title = {
					tier = prev.primary_title.tier
					title_capital_county.title_province = { geographical_region = btwk_world_britannia }
				}
				NOT = { title:e_britannia = { has_variable = anglo_saxons_defeated_all } }
			}
			text = conquest_flavour_briton.0002.a.anglosaxon
		}
		name = {
			trigger = {
				NOT = { this = scope:conqueror }
				culture = { 
					has_cultural_pillar = heritage_north_germanic
				}
				title:e_britannia = { has_variable = anglo_saxons_defeated_south }
			}
			text = conquest_flavour_briton.0002.a.norse_total_defeat
		}
		name = {
			trigger = {
				NOR = {
					culture = {
						OR = {
							has_cultural_pillar = heritage_brythonic
							has_cultural_pillar = heritage_latin
						}
					}
					AND = {
						NOT = { this = scope:conqueror }
						culture = { 
							has_cultural_pillar = heritage_west_germanic
						}
						any_held_title = {
							tier = prev.primary_title.tier
							title_capital_county.title_province = { geographical_region = btwk_world_britannia }
						}
					}
					AND = {
						NOT = { this = scope:conqueror }
						culture = { 
							has_cultural_pillar = heritage_north_germanic
						}
						title:e_britannia = { has_variable = anglo_saxons_defeated_all }
					}
				}
			}
			text = conquest_flavour_briton.0002.a
		}
	}
}
