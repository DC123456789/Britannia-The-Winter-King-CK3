# Toast for flipped provinces
spreading_of_christianity.0010 = {
	scope = none
	hidden = yes

	immediate = {
		scope:current_county.holder = {
			add_to_list = informees
			every_liege_or_above = {
				add_to_list = informees
			}
		}
		every_in_list = {
			list = informees
			switch = {
				trigger = faith
				scope:current_county.faith = { # Good if same as new faith
					send_interface_message = {
						type = msg_council_conversion_good
						title = spreading_of_christianity.0010.t
						desc = spreading_of_christianity.0010.desc
						right_icon = scope:current_county
					}
				}
				scope:old_faith = {
					send_interface_message = { # Bad if same faith as county
						type = msg_council_conversion_bad
						title = spreading_of_christianity.0010.t
						desc = spreading_of_christianity.0010.desc
						right_icon = scope:current_county
					}
				}
				fallback = {
					send_interface_message = { # Neutral otherwise
						type = msg_council_conversion_neutral
						title = spreading_of_christianity.0010.t
						desc = spreading_of_christianity.0010.desc
						right_icon = scope:current_county
					}
				}
			}
		}
	}
}

# Yearly check for if the mission should begin
spreading_of_christianity.0200 = {
	scope = none
	hidden = yes

	immediate = {
		if = {
			limit = {
				current_year > 590
				NOT = { situation:spreading_of_christianity_situation = { has_variable = gregorian_mission_started } }
				OR = {
					title:c_londinium = { has_clerical_region = no }
					title:c_lindum = { has_clerical_region = no }
				}
				any_county_in_region = {
					region = btwk_world_britannia
					percent >= 0.2
					holder.top_liege = {
						culture = { has_same_heritage_group_as = { TARGET = culture:angle } }
						faith = { NOT = { has_doctrine_parameter = unreformed } }
					}
				}
				random = {
					chance = {
						value = 0
						add = {
							value = current_year
							subtract = 590
							multiply = 2.5		# Around 52% chance of happening by the historical date (597)
						}
					}
				}
				trigger_event = spreading_of_christianity.0201
			}
		}
	}
}

# Mission begins, spawn Augustine of Canterbury and start the story
spreading_of_christianity.0201 = {
	scope = none
	hidden = yes

	immediate = {
		situation:spreading_of_christianity_situation = { set_variable = gregorian_mission_started }
		create_character = {
			age = 55		# Already a bit old when he arrives
			name = augustinus_male_roman
			dynasty = none
			gender_female_chance = 0
			random_traits = no
			location = title:c_durovernum.title_province
			faith = faith:christian_faith
			rite = rite:roman_rite		# Swaps to Saxon Rite after arrival
			culture = culture:roman
			trait = education_learning_4
			trait = devoted
			trait = erudite
			trait = arrogant
			trait = diligent
			trait = patient
			diplomacy = { min_template_decent_skill max_template_decent_skill }
			martial = { min_template_low_skill max_template_low_skill }
			stewardship = { min_template_high_skill max_template_high_skill }
			intrigue = { min_template_low_skill max_template_low_skill }
			learning = { min_template_high_skill max_template_high_skill }
			prowess = { min_template_low_skill max_template_low_skill }
			save_scope_as = historical_character
		}
		scope:historical_character ?= {
			add_trait_xp = {
				trait = devoted
				value = 75
			}
			add_piety = 750
			historical_character_finalization_effect = yes
			set_variable = BTWK_augustine_of_canterbury
			save_scope_as = augustine_of_canterbury
			add_character_flag = ai_will_not_convert
			create_story = {
				type = BTWK_soc_gregorian_mission_story
				save_scope_as = gregorian_mission_story
			}
			if = {
				limit = { exists = title:k_papal_state.holder }
				set_employer = title:k_papal_state.holder
			}
		}
		if = { limit = { has_variable = BTWK_augustine_of_canterbury  } } # Error suppression
		spawn_historical_characters_boilerplate_effect = yes
		
		# If they were invited by decision, save that character in the story, they will automatically go there
		if = {
			limit = { exists = scope:gregorian_mission_invited_ruler }
			scope:gregorian_mission_story = { 
				set_variable = { name = mission_host_king value = scope:gregorian_mission_invited_ruler }
			}
		}
	}
}

# Augustine asks King to establish mission + archbishopric in his realm
spreading_of_christianity.0202 = {
	type = character_event
	content_source = dlc_BTWK
	title = spreading_of_christianity.0202.t
	desc = spreading_of_christianity.0202.desc
	theme = faith
	override_background = { 
		reference = temple_church
	}
	
	left_portrait = {
		character = scope:gregorian_mission_augustine
		animation = writing
	}
	
	right_portrait = {
		character = root
		animation = idle
	}

	# Accept
	option = {
		name = spreading_of_christianity.0202.a
		custom_tooltip = spreading_of_christianity.0202.a.tt.1
		if = {
			limit = { highest_held_title_tier >= tier_kingdom }
			custom_tooltip = spreading_of_christianity.0202.a.tt.2
		}
		else = {
			custom_tooltip = spreading_of_christianity.0202.a.tt.2.not_king
		}
		custom_tooltip = spreading_of_christianity.0202.a.tt.3
		scope:gregorian_mission_story = {
			set_variable = { name = mission_host_king value = root }
		}
		
		# TODO Do AI weights - should generally accept unless zealous
		# Exception for when they are Bretwalda and being given priority - they should be less likely
		# to say yes in this case so Canterbury is the archbishopric center in most games
		
		ai_chance = {
			base = 100
			ai_value_modifier = {
				ai_zeal = -1.33
			}
		}
	}

	# Refuse
	option = {
		name = spreading_of_christianity.0202.b
		custom_tooltip = spreading_of_christianity.0202.b.tt
		if = {
			limit = {
				exists = scope:bretwalda_initial_request
				is_ai = yes
			}
			set_variable = bretwalda_initial_rejected_gregorian_mission
		}
		else = {
			set_variable = rejected_gregorian_mission
		}
		
		ai_chance = {
			base = -25
			ai_value_modifier = {
				ai_zeal = 1
			}
			# Baseline only ~50% chance of saying yes if initial Bretwalda request
			modifier = {
				add = 125
				exists = scope:bretwalda_initial_request
			}
			modifier = {
				add = {
					value = ai_rationality
					multiply = -0.5
				}
				ai_rationality >= 0
			}
		}
	}
}

# Mission arrives, Augustine arrives and establishes the archbishopric
# Saxon Rite spawns immediately when the mission arrives for simplicity
spreading_of_christianity.0203 = {
	type = character_event
	content_source = dlc_BTWK
	hidden = yes
	
	immediate = {		
		save_scope_as = gregorian_mission_augustine
		
		# Augustine abdicates any titles he happens to hold
		if = {
			limit = { is_ruler = yes }
			depose = yes
		}
		
		# Create the appropriate archbishopric
		save_scope_as = new_ecclesiastical_holder
		scope:mission_host_king = {
			random_in_global_list = {
				variable = creatable_clerical_title_stories
				# First, try the usual potential check - will work if they are Anglo-Saxon
				limit = {
					this.var:title = title:d_et_canterbury
					has_variable = anglo_saxon_kingdom_clerical_title
					save_temporary_scope_as = current_story
					prev = { 
						BTWK_potential_create_clerical_title_trigger = { 
							CLERICAL_TITLE_STORY = scope:current_story
						}
					}
				}
				# Then, try to fall back on wherever the capital of their primary title is
				alternative_limit = {
					this.var:title = title:d_et_canterbury
					NOT = { exists = this.var:seat.clerical_region_title }
					has_variable = anglo_saxon_kingdom_clerical_title
					prev.primary_title.title_capital_county.var:anglo_saxon_duchy.kingdom ?= this.anglo_saxon_kingdom_clerical_title
					this.var:seat.holder.top_liege = prev
					trigger_if = {
						limit = { has_variable = anglo_saxon_kingdom_specific_capital_clerical_title }
						this.var:anglo_saxon_kingdom_clerical_title.title_capital_county = this.var:anglo_saxon_kingdom_specific_capital_clerical_title
					}
					trigger_if = {
						limit = { has_variable = anglo_saxon_kingdom_not_specific_capital_clerical_title }
						this.var:anglo_saxon_kingdom_clerical_title.title_capital_county != this.var:anglo_saxon_kingdom_not_specific_capital_clerical_title
					}
				}
				# Then, try to fall back on any seat that they control
				alternative_limit = {
					this.var:title = title:d_et_canterbury
					NOT = { exists = this.var:seat.clerical_region_title }
					has_variable = anglo_saxon_kingdom_clerical_title
					this.var:seat.holder.top_liege = prev
					trigger_if = {
						limit = { has_variable = anglo_saxon_kingdom_specific_capital_clerical_title }
						this.var:anglo_saxon_kingdom_clerical_title.title_capital_county = this.var:anglo_saxon_kingdom_specific_capital_clerical_title
					}
					trigger_if = {
						limit = { has_variable = anglo_saxon_kingdom_not_specific_capital_clerical_title }
						this.var:anglo_saxon_kingdom_clerical_title.title_capital_county != this.var:anglo_saxon_kingdom_not_specific_capital_clerical_title
					}
				}
				# If that somehow breaks, then just default back to Canterbury
				alternative_limit = {
					this.var:title = title:d_et_canterbury
					NOT = { exists = this.var:seat.clerical_region_title }
					has_variable = no_southern_anglo_saxon_kingdom_clerical_title
				}
				# If that somehow *still* breaks, then any potential archbishopric which will work
				alternative_limit = {
					this.var:title = title:d_et_canterbury
					NOT = { exists = this.var:seat.clerical_region_title }
				}
				save_scope_as = canterbury_creation_story
			}
			scope:canterbury_creation_story = { set_variable = suppress_event }
			BTWK_create_clerical_title_effect = {
				CLERICAL_TITLE_STORY = scope:canterbury_creation_story
			}
		}
		
		# Start off with some Roman funds
		add_gold = 100
		add_treasury = 300
		
		# Augustine creates and converts to Saxon Rite
		create_rite_from_type = {
			type = saxon_rite
			convert = yes
		}
		set_character_rite = rite:saxon_rite
		
		# Any Roman Rite courtiers also convert to Saxon Rite
		every_courtier = {
			limit = { rite = rite:roman_rite }
			set_character_rite = rite:saxon_rite
		}
		
		# Spawn a Roman priests, some of whom will be monks
		while = {
			count = 5
			create_character = {
				location = this.capital_province
				template = priest_character_template
				faith = this.faith
				rite = this.rite
				culture = this.culture
				gender_female_chance = root_faith_clergy_gender_female_chance
				save_scope_as = new_priest
			}
			scope:new_priest = {
				set_employer = scope:gregorian_mission_augustine
				random = {
					chance = 50
					add_trait = devoted
					add_trait_xp = {
						trait = devoted
						value = 50
					}
				}
			}
		}
		
		# Spawn a few Frankish priests too
		while = {
			count = 3
			create_character = {
				location = this.capital_province
				template = priest_character_template
				faith = this.faith
				rite = rite:carolingian_christianity
				culture = culture:frankish
				gender_female_chance = root_faith_clergy_gender_female_chance
				save_scope_as = new_priest
			}
			scope:new_priest = {
				set_employer = scope:gregorian_mission_augustine
			}
		}
		
		# Notify everyone
		trigger_event = spreading_of_christianity.0204
		every_player = {
			limit = { this != scope:gregorian_mission_augustine }
			trigger_event = spreading_of_christianity.0204
		}
	}
}
