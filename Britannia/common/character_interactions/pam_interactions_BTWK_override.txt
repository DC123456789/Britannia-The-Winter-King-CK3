# Create a New Ecclesiastical Title as HoF (player only - we'll make the ai use the "request" ones)
create_new_ecclesiastical_title_interaction = {
	category = interaction_category_religion
	icon = request_clerical_title_creation
	interface_priority = 45
	common_interaction = yes

	use_diplomatic_range = yes
	ai_maybe = no
	popup_on_receive = yes

	desc = create_new_ecclesiastical_title_interaction_desc
	target_type = title
	target_filter = domain_titles
	target_source = recipient
	show_all_targets = no

	# Recipient's court chaplain as the default new title holder
	redirect = {
		scope:recipient = {
			# A character may hold only one clerical region, so an existing archbishop cannot default here
			cp:councillor_court_chaplain ?= {
				if = {
					limit = { NOT = { any_held_title = { has_clerical_region = yes } } }
					save_scope_as = secondary_recipient
				}
			}
		}
	}

	# Recipient's court chaplain, then eligible AI courtiers from either the recipient or the actor (HoF) as candidate holders
	populate_recipient_list = {
		scope:recipient = {
			cp:councillor_court_chaplain ?= {
				if = {
					limit = { NOT = { any_held_title = { has_clerical_region = yes } } }
					add_to_list = characters
				}
			}
			every_courtier = {
				limit = {
					is_ai = yes
					NOT = { any_held_title = { has_clerical_region = yes } }
					can_be_court_chaplain_trigger = { COURT_OWNER = scope:actor }
				}
				add_to_list = characters
			}
		}
		scope:actor = {
			every_courtier = {
				limit = {
					is_ai = yes
					NOT = { any_held_title = { has_clerical_region = yes } }
					can_be_court_chaplain_trigger = { COURT_OWNER = scope:actor }
				}
				add_to_list = characters
			}
		}
	}

	# Clerical region supporting HoF
	is_available = {
		always = no		# BTWK Disabled
		is_ai = no
		pam_is_clerical_title_authority_trigger = yes # HoF, or the Head of Rites when the faith has no HoF
		faith ?= { has_doctrine_parameter = has_clerical_regions }
		has_pam_dlc_trigger = yes
		is_adult = yes
	}

	# Same-faith duke-or-above holding a title whose de jure capital is not yet in a clerical region
	# We do not require the recipient's own capital, so a seated ruler can still found a see elsewhere
	is_shown = {
		scope:recipient = {
			is_ruler = yes
			faith ?= scope:actor.faith
			NOT = { this = scope:actor }
			any_held_title = {
				tier >= tier_duchy
				is_landless_type_title = no
				title_capital_county ?= {
					NOT = { exists = clerical_region_title }
					OR = {
						holder ?= scope:recipient
						holder ?= {
							any_liege_or_above = {
								this = scope:recipient
							}
						}
					}
				}
			}
		}
	}

	# At least one eligible clergy holder exists; at least one eligible kingdom title can be picked
	is_valid_showing_failures_only = {
		scope:actor = {
			is_imprisoned = no
			is_at_war = no
			is_incapable = no
		}
		scope:recipient = {
			is_imprisoned = no
			is_adult = yes
			pam_actor_is_clerical_title_authority_of_this_trigger = yes
		}
		custom_tooltip = {
			text = create_new_ecclesiastical_title_no_eligible_kingdom_tt
			scope:recipient = {
				any_held_title = {
					pam_valid_new_clerical_region_anchor_trigger = { PETITIONER = scope:recipient }
				}
			}
		}
		custom_tooltip = {
			text = create_new_ecclesiastical_title_no_eligible_holder_tt
			OR = {
				scope:recipient = {
					OR = {
						any_courtier = {
							can_be_court_chaplain_trigger = { COURT_OWNER = scope:recipient }
						}
						exists = cp:councillor_court_chaplain
					}
				}
				any_courtier = {
					can_be_court_chaplain_trigger = { COURT_OWNER = scope:actor }
				}
			}
		}
	}

	has_valid_target_showing_failures_only = {
		custom_tooltip = {
			text = pam_new_clerical_region_invalid_anchor_tt
			scope:target ?= {
				pam_valid_new_clerical_region_anchor_trigger = { PETITIONER = scope:recipient }
			}
		}
	}

	can_be_picked_title = {
		scope:target = {
			pam_valid_new_clerical_region_anchor_trigger = { PETITIONER = scope:recipient }
		}
	}

	# Scales with the free Christian de jure counties of the anchor's de jure kingdom, which is what the region takes
	cost = {
		piety = {
			value = pam_new_clerical_region_cost_value
			# A Head of Rites acting for a faith with no Head of Faith pays more
			if = {
				limit = { NOT = { exists = scope:actor.faith.religious_head } }
				multiply = 1.5
			}
		}
	}

	auto_accept = yes

	# Creates the new clerical region title anchored to the chosen kingdom, sets up the new holder as court chaplain, and fires the situation catalyst
	on_accept = {
		pam_create_new_ecclesiastical_title_interaction_effect = yes
	}
}

# Expand an Ecclesiastical Title as HoF (player only - we'll make the ai use the "request" ones)
expand_ecclesiastical_title_interaction = {
	category = interaction_category_religion
	icon = request_clerical_title_expansion
	interface_priority = 45
	common_interaction = no

	use_diplomatic_range = yes
	ai_maybe = no
	popup_on_receive = yes

	desc = expand_ecclesiastical_title_interaction_desc
	target_type = title
	target_filter = clerical_region_adjacent_duchies
	target_source = recipient
	show_all_targets = no

	# Clerical region supporting HoF
	is_available = {
		always = no		# BTWK Disabled
		is_ai = no
		pam_is_clerical_title_authority_trigger = yes # HoF, or the Head of Rites when the faith has no HoF
		faith ?= { has_doctrine_parameter = has_clerical_regions }
		has_pam_dlc_trigger = yes
		is_adult = yes
	}

	cooldown_against_recipient = { years = 2 }

	# Same-faith ecclesiastical rulers who hold a clerical region with at least one county bordering uncovered land; also the HoF on themselves
	is_shown = {
		scope:recipient = {
			is_ruler = yes
			faith ?= scope:actor.faith
			pam_government_is_religious_playable_trigger = yes
			# Picker only lists duchies around the region at the recipient's capital
			pam_acting_clerical_region_trigger = { ARCHBISHOP = this }
			pam_actor_is_clerical_title_authority_of_this_trigger = yes
		}
	}

	# At least one adjacent duchy passes the picker criteria
	is_valid_showing_failures_only = {
		scope:actor = {
			is_imprisoned = no
			is_at_war = no
			is_incapable = no
		}
		scope:recipient = {
			is_imprisoned = no
			is_adult = yes
			custom_tooltip = { # Must match can_be_picked_title below
				text = expand_ecclesiastical_title_no_eligible_target_tt
				pam_has_eligible_expand_ecclesiastical_title_target_trigger = yes
			}
		}
	}

	can_be_picked_title = {
		pam_valid_expand_ecclesiastical_title_target_trigger = { TITLE = scope:target ARCHBISHOP = scope:recipient }
	}

	# can_be_picked_title is picker-only and is not rechecked on send
	has_valid_target_showing_failures_only = {
		pam_valid_expand_ecclesiastical_title_target_trigger = { TITLE = scope:target ARCHBISHOP = scope:recipient }
	}

	# Piety scales with the number of de jure counties in the chosen target duchy
	cost = {
		piety = {
			value = pam_new_ecclesiastical_title_cost_value
			# A Head of Rites acting for a faith with no Head of Faith pays more
			if = {
				limit = { NOT = { exists = scope:actor.faith.religious_head } }
				multiply = 1.5
			}
		}
	}

	auto_accept = yes

	# Grows the recipient's clerical region into all de jure counties of the chosen duchy and fires the situation catalyst
	on_accept = {
		pam_expand_ecclesiastical_title_interaction_effect = yes
	}
}

# Split an Ecclesiastical Title as HoF (player only - we'll make the ai use the "request" ones)
split_ecclesiastical_title_interaction = {
	category = interaction_category_religion
	icon = request_clerical_title_split
	interface_priority = 45
	common_interaction = no

	use_diplomatic_range = yes
	ai_maybe = no
	popup_on_receive = yes

	desc = split_ecclesiastical_title_interaction_desc

	# Pre-selects a default secular ruler (secondary_actor) by finding a random eligible county holder or liege in the recipient's clerical region THEN pre-selects their court chaplain as the default new holder (secondary_recipient) if available
	redirect = {
		scope:recipient = {
			random_held_title = {
				limit = {
					has_clerical_region = yes
					is_head_of_faith = no
				}
				random_county_in_clerical_region = {
					limit = {
						OR = {
							holder = {
								pam_potential_secular_clerical_split_ruler_trigger = { RULER = this }
							}
							holder = {
								any_liege_or_above = {
									pam_potential_secular_clerical_split_ruler_trigger = { RULER = this }
								}
							}
						}
					}
					holder = {
						if = {
							limit = {
								pam_potential_secular_clerical_split_ruler_trigger = { RULER = this }
							}
							save_scope_as = secondary_actor
						}
						else = {
							random_liege_or_above = {
								limit = {
									pam_potential_secular_clerical_split_ruler_trigger = { RULER = this }
								}
								save_scope_as = secondary_actor
							}
						}
					}
				}
			}
		}
		if = {
			limit = {
				scope:secondary_actor ?= {
					is_ai = yes
					exists = cp:councillor_court_chaplain
					NOT = { cp:councillor_court_chaplain = scope:recipient }
				}
			}
			scope:secondary_actor = {
				cp:councillor_court_chaplain = {
					save_scope_as = secondary_recipient
				}
			}
		}
	}

	# Lists all eligible duke+ secular rulers whose realm capital lies within the recipient's clerical region, found by county holders and their liege chains
	populate_actor_list = {
		scope:recipient = {
			every_held_title = {
				limit = {
					has_clerical_region = yes
					is_head_of_faith = no
				}
				every_county_in_clerical_region = {
					holder = {
						if = {
							limit = {
								pam_potential_secular_clerical_split_ruler_trigger = { RULER = this }
							}
							add_to_list = characters
						}
						every_liege_or_above = {
							limit = {
								pam_potential_secular_clerical_split_ruler_trigger = { RULER = this }
							}
							add_to_list = characters
						}
					}
				}
			}
		}
	}

	# Lists clergy candidates for the new title: the chosen secular ruler's court chaplain (if eligible), their eligible courtiers, and same-faith ecclesiastical holders/top lieges of counties within the recipient's clerical region - excludes existing clerical region holders
	populate_recipient_list = {
		if = {
			limit = { exists = scope:secondary_actor }
			scope:secondary_actor = {
				if = {
					limit = {
						exists = cp:councillor_court_chaplain
						cp:councillor_court_chaplain != scope:recipient
						NOT = {
							cp:councillor_court_chaplain = {
								any_held_title = { has_clerical_region = yes }
							}
						}
					}
					cp:councillor_court_chaplain = { add_to_list = characters }
				}
				every_courtier = {
					limit = {
						is_ai = yes
						this != scope:recipient
						NOT = { any_held_title = { has_clerical_region = no } }
						can_be_court_chaplain_trigger = { COURT_OWNER = scope:secondary_actor }
					}
					add_to_list = characters
				}
			}
		}
		scope:recipient = {
			primary_title = {
				every_county_in_clerical_region = {
					if = {
						limit = {
							holder = {
								is_alive = yes
								is_ruler = yes
								faith = scope:actor.faith
								pam_government_is_religious_playable_trigger = yes
								this != scope:recipient
								NOT = { any_held_title = { has_clerical_region = yes } }
							}
						}
						holder = {
							add_to_list = characters
						}
					}
					if = {
						limit = {
							holder.top_liege = {
								is_alive = yes
								is_ruler = yes
								faith = scope:actor.faith
								pam_government_is_religious_playable_trigger = yes
								this != scope:recipient
								NOT = { any_held_title = { has_clerical_region = yes } }
							}
							holder != scope:recipient
						}
						holder.top_liege = {
							add_to_list = characters
						}
					}
				}
			}
		}
	}

	# Clerical region supporting HoF
	is_available = {
		always = no		# BTWK Disabled
		is_ai = no
		pam_is_clerical_title_authority_trigger = yes # HoF, or the Head of Rites when the faith has no HoF
		faith ?= { has_doctrine_parameter = has_clerical_regions }
		has_pam_dlc_trigger = yes
		is_adult = yes
	}

	cooldown_against_recipient = { years = 10 }

	# Shown for same-faith ecclesiastical rulers with at least one eligible secular ruler inside it; also the HoF on their own region
	is_shown = {
		scope:recipient = {
			is_ruler = yes
			faith ?= scope:actor.faith
			pam_government_is_religious_playable_trigger = yes
			any_held_title = {
				has_clerical_region = yes
				# The HoF may split their own see; other rulers' head-of-faith titles stay protected
				trigger_if = {
					limit = { scope:recipient != scope:actor }
					is_head_of_faith = no
				}
				any_county_in_clerical_region = {
					OR = {
						holder ?= {
							pam_potential_secular_clerical_split_ruler_trigger = { RULER = this }
						}
						holder ?= {
							any_liege_or_above = {
								pam_potential_secular_clerical_split_ruler_trigger = { RULER = this }
							}
						}
					}
				}
			}
		}
	}

	# A valid secular ruler exists to split from, and that an eligible clergy holder candidate exists
	is_valid_showing_failures_only = {
		scope:actor = {
			is_imprisoned = no
			is_at_war = no
			is_incapable = no
		}
		scope:recipient = {
			is_imprisoned = no
			is_adult = yes
			pam_actor_is_clerical_title_authority_of_this_trigger = yes
		}
		custom_tooltip = {
			text = split_ecclesiastical_title_no_eligible_secular_ruler_tt
			scope:recipient = {
				any_held_title = {
					has_clerical_region = yes
					# Mirror is_shown: the HoF may split their own see, other rulers' head-of-faith titles stay protected
					trigger_if = {
						limit = { scope:recipient != scope:actor }
						is_head_of_faith = no
					}
					any_county_in_clerical_region = {
						OR = {
							holder = {
								pam_valid_secular_clerical_split_ruler_trigger = { RULER = this }
							}
							holder = {
								any_liege_or_above = {
									pam_valid_secular_clerical_split_ruler_trigger = { RULER = this }
								}
							}
						}
					}
				}
			}
			trigger_if = {
				limit = { exists = scope:secondary_actor }
				pam_valid_secular_clerical_split_ruler_trigger = { RULER = scope:secondary_actor }
			}
		}
		custom_tooltip = {
			text = split_ecclesiastical_title_no_eligible_holder_tt
			OR = {
				AND = {
					exists = scope:secondary_actor
					OR = {
						scope:secondary_actor = {
							exists = cp:councillor_court_chaplain
							cp:councillor_court_chaplain != scope:recipient
						}
						scope:secondary_actor = {
							any_courtier = {
								this != scope:recipient
								can_be_court_chaplain_trigger = { COURT_OWNER = scope:secondary_actor }
							}
						}
					scope:recipient = {
						primary_title = {
							any_county_in_clerical_region = {
								holder = {
									is_alive = yes
									is_ruler = yes
									faith = scope:actor.faith
									pam_government_is_religious_playable_trigger = yes
									this != scope:recipient
									NOT = { any_held_title = { has_clerical_region = yes } }
								}
							}
						}
					}
					}
				}
				NOT = { exists = scope:secondary_actor }
			}
		}
	}

	# Piety scales with the number of de jure counties in the chosen secular ruler's primary title
	cost = {
		piety = {
			value = pam_split_ecclesiastical_title_cost_value
			# A Head of Rites acting for a faith with no Head of Faith pays more
			if = {
				limit = { NOT = { exists = scope:actor.faith.religious_head } }
				multiply = 1.5
			}
		}
	}

	auto_accept = yes

	# Splits the secular ruler's land out of the clerical region, creates the new title, grants it to the chosen holder, and fires the situation catalyst
	on_accept = {
		pam_split_ecclesiastical_title_interaction_effect = yes
	}
}

# Merge an Ecclesiastical Title as HoF (player only - we'll make the ai use the "request" ones)
merge_new_ecclesiastical_title_interaction = {
	category = interaction_category_religion
	icon = request_clerical_title_merger
	interface_priority = 45
	common_interaction = no

	use_diplomatic_range = yes
	ai_maybe = no
	popup_on_receive = yes

	desc = merge_new_ecclesiastical_title_interaction_desc
	target_type = title
	target_filter = clerical_region
	show_all_targets = no

	# Clerical region supporting HoF
	is_available = {
		always = no		# BTWK Disabled
		is_ai = no
		pam_is_clerical_title_authority_trigger = yes # HoF, or the Head of Rites when the faith has no HoF
		faith ?= { has_doctrine_parameter = has_clerical_regions }
		has_pam_dlc_trigger = yes
		is_adult = yes
	}

	cooldown_against_recipient = { years = 20 }

	# Same-faith ecclesiastical rulers that border at least one other same-faith clerical region; also the HoF on themselves
	is_shown = {
		scope:recipient = {
			is_ruler = yes
			faith ?= scope:actor.faith
			pam_government_is_religious_playable_trigger = yes
			any_held_title = {
				has_clerical_region = yes
				is_head_of_faith = no
			}
			pam_actor_is_clerical_title_authority_of_this_trigger = yes
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			is_imprisoned = no
			is_at_war = no
			is_incapable = no
		}
		scope:recipient = {
			is_imprisoned = no
			is_adult = yes
			custom_tooltip = {
				text = merge_new_ecclesiastical_title_cooldown_trigger
				NOT = { exists = capital_province.county.clerical_region_title.var:merge_clerical_region_cooldown }
			}
			# Must be the recipient's own region, because a landed archbishop's capital can sit in another region
			pam_acting_clerical_region_trigger = { ARCHBISHOP = this }
			capital_province.county.clerical_region_title ?= {
				pam_has_adjacent_mergeable_clerical_region_trigger = yes
			}
		}
	}

	can_be_picked_title = {
		pam_acting_clerical_region_trigger = { ARCHBISHOP = scope:recipient }
		pam_potential_merge_clerical_region_title_trigger = {
			TITLE = scope:target
			ABSORBING_TITLE = scope:recipient.capital_province.county.clerical_region_title
		}
	}

	# can_be_picked_title is picker-only and is not rechecked on send
	has_valid_target_showing_failures_only = {
		pam_acting_clerical_region_trigger = { ARCHBISHOP = scope:recipient }
		pam_potential_merge_clerical_region_title_trigger = {
			TITLE = scope:target
			ABSORBING_TITLE = scope:recipient.capital_province.county.clerical_region_title
		}
	}

	# Piety scales with the number of counties in the absorbed clerical region
	cost = {
		piety = {
			value = pam_merge_new_ecclesiastical_title_cost_value
			# A Head of Rites acting for a faith with no Head of Faith pays more
			if = {
				limit = { NOT = { exists = scope:actor.faith.religious_head } }
				multiply = 2
			}
		}
	}

	auto_accept = yes

	# Absorbs all counties of the target clerical region into the recipient's primary title via grow_clerical_region_fromand destroys it
	on_accept = {
		pam_merge_new_ecclesiastical_title_interaction_effect = {
			ABSORBING_TITLE = scope:recipient.capital_province.county.clerical_region_title
			ABSORBED_TITLE = scope:target
			SENDER = scope:actor
		}
	}
}

petition_antipope_indulgences_interaction = {
	category = interaction_category_religion
	icon = petition_antipope_indulgences
	common_interaction = yes

	desc = petition_antipope_indulgences_interaction_desc
	notification_text = RELIGIOUS_HEAD_SEEK_INDULGENCES
	
	ai_targets = {
		ai_recipients = religious_head_challenger
	}
	ai_frequency_by_tier = {
		barony = 0
		county = 36
		duchy = 24
		kingdom = 12
		empire = 12
		hegemony = 12
	}

	is_available = {
		always = no		# BTWK Disabled
		faith = {
			faith_has_sacraments_central_trigger = yes
			OR = {
				has_doctrine_parameter = has_clerical_regions
				has_doctrine = doctrine_spiritual_head
				has_doctrine = doctrine_temporal_head
			}
		}
		rite = {
			NOT = {
				rite_has_parameter = no_seek_indulgences
			}
		}
	}

	is_shown = {
		# Anyone under an antipope or their sponsor petitions that antipope, not the Head of Faith
		scope:actor = { is_following_religious_head_challenger = yes }
		has_pam_dlc_trigger = yes
		scope:actor != scope:recipient
		scope:actor.religious_head_or_challenger ?= scope:recipient
	}

	cooldown = { years = 5 }

	is_valid_showing_failures_only = {
		scope:actor = {
			NOT = {
				is_at_war_with = scope:recipient
			}
			NOT = {
				has_character_modifier = promised_pilgrimage_to_clergy_modifier
			}
			trigger_if = {
				limit = {
					has_trait = kinslayer_3
					trait_is_shunned_or_criminal_in_rite_trigger = { RITE = rite TRAIT = trait:kinslayer_3 GENDER_CHARACTER = scope:actor }
				}
				gold >= seek_indulgences_major_criminal_trait_value
			}
			trigger_else_if = {
				limit = {
					OR = {

							this = scope:puppet_or_actor
							any_liege_or_above = {
								this = scope:puppet_or_actor
							}
						}
					}
				}
			}
			add = 50
		}
		modifier = {
			OR = {
				ai_greed < 0
				ai_energy < 0
				AND = {
					scope:target.de_jure_liege.holder ?= {
						NOR = {
							this = scope:puppet_or_actor
							this = scope:recipient
						}
					}
					has_any_good_relationship_with_character_trigger = { CHARACTER = scope:target.de_jure_liege.holder }
				}
				trigger_if = {
					limit = {
						has_legitimacy = yes
					}
					legitimacy_level < 2
				}
			}
			factor = 0
		}
	}
