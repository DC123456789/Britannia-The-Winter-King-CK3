BTWK_soc_gregorian_mission_story = {
	visible = yes
	
	icon = {
		trigger = {
			always = yes
		}
		reference = "gfx/interface/icons/the_christian_church/phase_type_icons/conversion.dds"
	}

	on_setup = {
	}

	on_end = {
		if = {
			limit = { NOT = { has_variable = mission_seat_selected } }
			situation:spreading_of_christianity_situation = { set_variable = gregorian_mission_failed }
			every_player = {
				limit = { 
					OR = {
						capital_province = { geographical_region = btwk_world_britannia }
						is_britannic_culture_trigger = yes
						is_anglo_saxon_culture_trigger = yes
					}
				}
				trigger_event = spreading_of_christianity.0205
			}
		}
	}

	on_owner_death = {
		# TODO Check if this actually works
		# If the Archbishopric of Canterbury exists and has a holder, transfer it over to them
		# Otherwise, create a random new priest character and transfer it to them
		if = {
			limit = { exists = title:d_et_canterbury.var:clerical_title.holder }
			make_story_owner = title:d_et_canterbury.var:clerical_title.holder
		}
		else = {
			create_character = {
				location = story_owner.capital_province
				template = priest_character_template
				faith = story_owner.faith
				rite = story_owner.rite
				culture = story_owner.culture
				gender_female_chance = scope_actor_faith_clergy_gender_female_chance
				save_scope_as = new_mission_head
			}
			make_story_owner = scope:new_mission_head
		}
	}
	
	# Initial cycle for asking Anglo-Saxon kings if they can come and do missionary work in their kingdom
	# Will always ask top liege, will not ask again if they rejected, and the capital of the kingdom
	# must not already be in a clerical region
	effect_group = {
		days = 15
		
		trigger = {
			NOT = { has_variable = mission_host_king }
			NOT = { has_variable = mission_started }
		}

		first_valid = {
			# If invited by a Bretwalda, will go found the archbishopric in their capital
			# No need to ask for permission, you were invited!
			triggered_effect = {
				trigger = {
					this.var:mission_host_king ?= {
						is_alive = yes
						is_independent_ruler = yes
						any_held_title = {
							tier >= tier_kingdom
							title_capital_county.title_province = { geographical_region = btwk_world_britannia_south }
							title_capital_county = { has_clerical_region = no }
						}
						any_realm_county = {
							title_province = { geographical_region = btwk_world_britannia_south }
						}
					}
				}
				effect = {
					this.var:mission_host_king = { save_scope_as = mission_host_king }
					story_owner = { trigger_event = spreading_of_christianity.0203 }
				}
			}
			# If there exists a Bretwalda, will always approach them first (?)
			# 50% chance of AI rejecting here, since we want Canterbury to happen most of the time
			triggered_effect = {
				trigger = {
					title:e_aengland = {
						has_variable = aengland_primary_kingdom
						this.var:aengland_primary_kingdom.title_capital_county.title_province = { geographical_region = btwk_world_britannia_south }
						title_capital_county = { has_clerical_region = no }
					}
					title:e_aengland.holder ?= {
						is_independent_ruler = yes
						NOT = { has_variable = bretwalda_initial_rejected_gregorian_mission }
						NOT = { has_variable = rejected_gregorian_mission }
						any_realm_county = {
							title_province = { geographical_region = btwk_world_britannia_south }
						}
					}
				}
				effect = {
					save_scope_as = gregorian_mission_story
					save_scope_as = bretwalda_initial_request
					story_owner = { save_scope_as = gregorian_mission_augustine }
					title:e_aengland.holder = { trigger_event = spreading_of_christianity.0202 }
				}
			}
			# Otherwise, ask the King of Kent
			triggered_effect = {
				trigger = {
					title:k_cantia.holder ?= {
						is_independent_ruler = yes
						NOT = { has_variable = rejected_gregorian_mission }
						title_capital_county = { has_clerical_region = no }
						any_realm_county = {
							title_province = { geographical_region = btwk_world_britannia_south }
						}
					}
				}
				effect = {
					save_scope_as = gregorian_mission_story
					story_owner = { save_scope_as = gregorian_mission_augustine }
					random_in_global_list = {
						variable = creatable_clerical_title_stories
						limit = {
							this.var:title = title:d_et_canterbury
							this.var:seat = title:c_durovernum
						}
						save_scope_as = mission_archdiocese_story
					}
					title:k_cantia.holder = { trigger_event = spreading_of_christianity.0202 }
				}
			}
			# Otherwise, ask the owner of Canterbury if Anglo-Saxon
			triggered_effect = {
				trigger = {
					title:c_durovernum = { has_clerical_region = no }
					title:c_durovernum.holder.top_liege ?= {
						is_unromanized_germanic_culture_trigger = yes
						NOT = { has_variable = rejected_gregorian_mission }
					}
				}
				effect = {
					save_scope_as = gregorian_mission_story
					story_owner = { save_scope_as = gregorian_mission_augustine }
					random_in_global_list = {
						variable = creatable_clerical_title_stories
						limit = {
							this.var:title = title:d_et_canterbury
							this.var:seat = title:c_durovernum
						}
						save_scope_as = mission_archdiocese_story
					}
					title:c_durovernum.holder.top_liege = { trigger_event = spreading_of_christianity.0202 }
				}
			}
			# Otherwise, ask the King of Essex
			triggered_effect = {
				trigger = {
					title:k_eastseaxe.holder ?= {
						is_independent_ruler = yes
						NOT = { has_variable = rejected_gregorian_mission }
						title_capital_county = { has_clerical_region = no }
						any_realm_county = {
							title_province = { geographical_region = btwk_world_britannia_south }
						}
					}
				}
				effect = {
					save_scope_as = gregorian_mission_story
					story_owner = { save_scope_as = gregorian_mission_augustine }
					random_in_global_list = {
						variable = creatable_clerical_title_stories
						limit = {
							this.var:title = title:d_et_canterbury
							this.var:seat = title:c_londinium
						}
						save_scope_as = mission_archdiocese_story
					}
					title:k_eastseaxe.holder = { trigger_event = spreading_of_christianity.0202 }
				}
			}
			# Otherwise, ask the owner of London if Anglo-Saxon
			triggered_effect = {
				trigger = {
					title:c_londinium = { has_clerical_region = no }
					title:c_londinium.holder.top_liege ?= {
						is_unromanized_germanic_culture_trigger = yes
						NOT = { has_variable = rejected_gregorian_mission }
					}
				}
				effect = {
					save_scope_as = gregorian_mission_story
					story_owner = { save_scope_as = gregorian_mission_augustine }
					random_in_global_list = {
						variable = creatable_clerical_title_stories
						limit = {
							this.var:title = title:d_et_canterbury
							this.var:seat = title:c_londinium
						}
						save_scope_as = mission_archdiocese_story
					}
					title:c_londinium.holder.top_liege = { trigger_event = spreading_of_christianity.0202 }
				}
			}
			# Otherwise, go to each Anglo-Saxon king in order of realm size
			triggered_effect = {
				trigger = {
					any_independent_ruler = {
						NOT = { has_variable = rejected_gregorian_mission }
						is_unromanized_germanic_culture_trigger = yes
						any_held_title = {
							tier >= tier_kingdom
							tier = prev.primary_title.tier
							title_capital_county.title_province = { 
								geographical_region = btwk_world_britannia_south 
							}
							title_capital_county = { has_clerical_region = no }
						}
						any_realm_county = {
							title_province = { geographical_region = btwk_world_britannia_south }
						}
					}
				}
				effect = {
					save_scope_as = gregorian_mission_story
					story_owner = { save_scope_as = gregorian_mission_augustine }
					ordered_independent_ruler = {
						limit = {
							NOT = { has_variable = rejected_gregorian_mission }
							is_unromanized_germanic_culture_trigger = yes
							any_held_title = {
								tier >= tier_kingdom
								tier = prev.primary_title.tier
								title_capital_county.title_province = { 
									geographical_region = btwk_world_britannia_south 
								}
								title_capital_county = { has_clerical_region = no }
							}
							any_realm_county = {
								title_province = { geographical_region = btwk_world_britannia_south }
							}
						}
						order_by = sub_realm_size
						random_in_global_list = {
							variable = creatable_clerical_title_stories
							limit = {
								this.var:title = title:d_et_canterbury
								has_variable = anglo_saxon_kingdom_clerical_title
								save_temporary_scope_as = current_story
								prev = { 
									BTWK_potential_create_clerical_title_trigger = { 
										CLERICAL_TITLE_STORY = scope:current_story
									}
								}
							}
							alternative_limit = {
								this.var:title = title:d_et_canterbury
								has_variable = anglo_saxon_kingdom_clerical_title
								this.var:anglo_saxon_kingdom_clerical_title.holder = prev
							}
							save_scope_as = mission_archdiocese_story
						}
						trigger_event = spreading_of_christianity.0202
					}
				}
			}
			# Otherwise, go to potential seat holders in order of realm size
			# Anglo-Saxon rulers are prioritized
			triggered_effect = {
				trigger = {
					any_in_global_list = {
						variable = creatable_clerical_title_stories
						this.var:title = title:d_et_canterbury
						this.var:seat.holder.top_liege = {
							NOT = { has_variable = rejected_gregorian_mission }
						}
						this.var:seat = { has_clerical_region = no }
						trigger_if = {
							limit = { has_variable = anglo_saxon_kingdom_clerical_title }
							this.var:anglo_saxon_kingdom_clerical_title = {
								any_direct_de_jure_vassal_title = { count >= 1 }
							}
							trigger_if = {
								limit = { has_variable = anglo_saxon_kingdom_specific_capital_clerical_title }
								this.var:anglo_saxon_kingdom_clerical_title.title_capital_county = this.var:anglo_saxon_kingdom_specific_capital_clerical_title
							}
							trigger_if = {
								limit = { has_variable = anglo_saxon_kingdom_not_specific_capital_clerical_title }
								this.var:anglo_saxon_kingdom_clerical_title.title_capital_county != this.var:anglo_saxon_kingdom_not_specific_capital_clerical_title
							}
						}
					}
				}
				effect = {
					save_scope_as = gregorian_mission_story
					story_owner = { save_scope_as = gregorian_mission_augustine }
					ordered_in_global_list = {
						variable = creatable_clerical_title_stories
						limit = {
							this.var:seat.holder.top_liege = {
								NOT = { has_variable = rejected_gregorian_mission }
							}
							this.var:seat = { has_clerical_region = no }
							trigger_if = {
								limit = { has_variable = anglo_saxon_kingdom_clerical_title }
								this.var:anglo_saxon_kingdom_clerical_title = {
									any_direct_de_jure_vassal_title = { count >= 1 }
								}
								trigger_if = {
									limit = { has_variable = anglo_saxon_kingdom_specific_capital_clerical_title }
									this.var:anglo_saxon_kingdom_clerical_title.title_capital_county = this.var:anglo_saxon_kingdom_specific_capital_clerical_title
								}
								trigger_if = {
									limit = { has_variable = anglo_saxon_kingdom_not_specific_capital_clerical_title }
									this.var:anglo_saxon_kingdom_clerical_title.title_capital_county != this.var:anglo_saxon_kingdom_not_specific_capital_clerical_title
								}
							}
						}
						order_by = {
							value = this.var:seat.holder.top_liege.sub_realm_size
							if = {
								limit = { 
									this.var:seat.holder.top_liege = {
										is_unromanized_germanic_culture_trigger = yes
									}
								}
								add = 20000
							}
						}
						save_scope_as = mission_archdiocese_story
						this.var:seat.holder.top_liege = { trigger_event = spreading_of_christianity.0202 }
					}
				}
			}
		}
		
		fallback = {
			# End mission if there is no one eligible left who will accept the mission
			end_story = yes
		}
	}
	
	# Once the request is accepted, they missionaries arrive one month later
	# Need to make sure that the king didn't die or get deposed in the meantime though
	effect_group = {
		months = 1
		
		trigger = {
			has_variable = mission_host_king
			NOT = { has_variable = mission_started }
		}

		first_valid = {
			# If the king is still capable of accepting the mission, the mission begins
			triggered_effect = {
				trigger = {
					this.var:mission_host_king = {
						is_alive = yes
						is_independent_ruler = yes
						NOT = { has_variable = rejected_gregorian_mission }
						OR = {
							any_held_title = {
								tier >= tier_kingdom
								title_capital_county.title_province = { geographical_region = btwk_world_britannia_south }
								title_capital_county = { has_clerical_region = no }
							}
							any_in_global_list = {
								variable = creatable_clerical_title_stories
								this.var:title = title:d_et_canterbury
								this.var:seat.holder.top_liege = prev
								this.var:seat = { has_clerical_region = no }
							}
						}
						any_realm_county = {
							title_province = { geographical_region = btwk_world_britannia_south }
						}
					}
				}
				effect = {
					this.var:mission_host_king = { save_scope_as = mission_host_king }
					story_owner = { trigger_event = spreading_of_christianity.0203 }
					set_variable = mission_started
				}
			}
		}
		
		fallback = {
			# If the chosen king is no longer valid, then we go back to looking for kings to accept the mission
			remove_variable = mission_host_king
		}
	}
}
