			type = creatable_clerical_title_story
			save_temporary_scope_as = new_creatable_clerical_title_story

		scope:new_creatable_clerical_title_story = {
			set_variable = { name = title value = $TITLE$ }
			set_variable = { name = seat value = $SEAT$ }

				name = creatable_clerical_title_stories
				target = THIS
			}
		}
	}
}

BTWK_create_clerical_title_effect = {
	save_scope_as = kingdom_holder
	save_scope_as = actor
	save_scope_as = puppet_or_actor
	$CLERICAL_TITLE_STORY$ = { save_scope_as = clerical_title_creation_story }
	$CLERICAL_TITLE_STORY$.var:region = { save_scope_as = effect_region }
	$CLERICAL_TITLE_STORY$.var:title = { save_scope_as = clerical_title_template }
	$CLERICAL_TITLE_STORY$.var:coa_title = { save_scope_as = clerical_title_coa_template }
	$CLERICAL_TITLE_STORY$.var:seat = { save_scope_as = clerical_title_seat }
	if = {
		limit = { $CLERICAL_TITLE_STORY$ = { has_variable = colour_title } }
		$CLERICAL_TITLE_STORY$.var:colour_title = { save_scope_as = clerical_title_colour_template }
	}
	else = {
		$CLERICAL_TITLE_STORY$.var:title = { save_scope_as = clerical_title_colour_template }
	}
	# Get the best candidate for the new archbishop, unless someone has already been selected
	if = {
		limit = { NOT = { exists = scope:new_ecclesiastical_holder } }
		if = {
			limit = { 
				has_government = ecclesiastical_government
				primary_title.title_capital_county = {
					title_province = { geographical_region = scope:effect_region }
					NOT = { exists = clerical_region_title }
				}
				NOT = { any_held_title = { has_clerical_region = yes } }
			}
			save_scope_as = new_ecclesiastical_holder
		}
		else_if = {
			limit = {
				primary_title.title_capital_county = {
					title_province = { geographical_region = scope:effect_region }
					NOT = { exists = clerical_region_title }
				}
				exists = cp:councillor_court_chaplain
				cp:councillor_court_chaplain = {
					faith = scope:kingdom_holder.faith
					NOT = { any_held_title = { has_clerical_region = yes } }
				}
			}
			cp:councillor_court_chaplain = {
				save_scope_as = new_ecclesiastical_holder
			}
		}
		else_if = {
			limit = {
				any_vassal = {
					primary_title.title_capital_county = {
						title_province = { geographical_region = scope:effect_region }
						NOT = { exists = clerical_region_title }
					}
					faith = scope:kingdom_holder.faith
					has_government = ecclesiastical_government
				}
			}
			ordered_vassal = {
				limit = {
					primary_title.title_capital_county = {
						title_province = { geographical_region = scope:effect_region }
						NOT = { exists = clerical_region_title }
					}
					faith = scope:kingdom_holder.faith
					has_government = ecclesiastical_government
				}
				order_by = learning
				save_scope_as = new_ecclesiastical_holder
			}
		}
		else = {
			ordered_courtier = {
				limit = {
					NOT = { any_held_title = { has_clerical_region = yes } }
					faith = scope:kingdom_holder.faith
					can_be_court_chaplain_trigger = { COURT_OWNER = scope:kingdom_holder }
				}
				order_by = learning
				save_scope_as = new_ecclesiastical_holder
			}
		}
	}
	if = {
		limit = { exists = scope:new_ecclesiastical_holder }
		custom_tooltip = BTWK_create_clerical_title_effect_create_archbishopric_tt
	}
	else = {
		custom_tooltip = BTWK_create_clerical_title_effect_create_archbishopric_no_existing_holder_tt
		create_character = {
			location = this.capital_province
			template = priest_character_template
			faith = this.faith
			rite = this.rite
			culture = this.culture
			gender_female_chance = scope_actor_faith_clergy_gender_female_chance
			save_scope_as = new_ecclesiastical_holder
		}
	}
	# We bring the new holder into court as chaplain unless they already hold the seat
	if = {
		limit = { 
			scope:kingdom_holder != scope:new_ecclesiastical_holder
			NOT = { exists = primary_title.title_capital_county.clerical_region_title }
			primary_title.title_capital_county.title_province = {
				geographical_region = scope:effect_region
			}
		}
		scope:kingdom_holder = {
			if = {
				limit = {
					scope:new_ecclesiastical_holder = {
						NOT = { is_courtier_of = scope:kingdom_holder }
						is_ruler = no
					}
				}
				add_courtier = scope:new_ecclesiastical_holder
			}
			if = {
				limit = {
					scope:new_ecclesiastical_holder = {
						NOT = { is_councillor_of = scope:kingdom_holder }
					}
				}
				assign_councillor_type = {
					type = councillor_court_chaplain
					remove_existing_councillor = yes
					target = scope:new_ecclesiastical_holder
				}
			}
		}
	}
	# If the new archbishop was not previously landed, they will become your vassal
	if = {
		limit = { scope:new_ecclesiastical_holder ?= { is_ruler = no } }
		save_scope_as = vassalize_new_ecclesiastical_holder
	}
	# Actually create the title
	$CLERICAL_TITLE_STORY$ = {
		set_variable = clerical_title_created
		scope:clerical_title_seat.title_province = {
			create_clerical_region = {
				domicile_type = ecclesiastical_domicile
				name = CREATED_CLERICAL_TITLE_NAME
				save_scope_as = new_title
			}
		}
		create_title_and_vassal_change = {
			type = created
			save_scope_as = change
			add_claim_on_loss = no
		}
		scope:new_title ?= {
			set_coa = scope:clerical_title_coa_template
			set_color_from_title = scope:clerical_title_colour_template
			set_capital_county = scope:clerical_title_seat
			set_landless_title = yes
			set_no_automatic_claims = yes
			set_always_follows_primary_heir = yes
			change_title_holder = {
				holder = scope:new_ecclesiastical_holder
				change = scope:change
			}
			if = {
				limit = { exists = scope:vassalize_new_ecclesiastical_holder }
				scope:new_ecclesiastical_holder = {
					change_liege = {
						liege = scope:kingdom_holder
						change = scope:change
					}
				}
			}
			resolve_title_and_vassal_change = scope:change
		}
		scope:clerical_title_template = {
			set_variable = { name = clerical_title value = scope:new_title }
		}
		hidden_effect = {
			# Need a while loop, as only counties *currently* adjacent to the 
			# archbishopric will be added in each pass
			while = {
				limit = {
					any_county_in_region = {
						region = scope:effect_region
						NOT = { exists = clerical_region_title }
						# Prevent maxing out the while loop for isolated islands, until this gets changed by Paradox
						any_neighboring_county = { clerical_region_title = scope:new_title }
					}
				}
				every_county_in_region = {
					region = scope:effect_region
					limit = { NOT = { exists = clerical_region_title } }
					scope:new_title = { expand_clerical_region = prev }
				}
			}
		}
	}
	hidden_effect = {
		if = {
			limit = {
				has_pam_dlc_trigger = yes
				religion = religion:christianity_religion
				any_character_situation = {
					situation_type = the_christian_church
					situation_top_has_catalyst = catalyst_the_christian_church_clerical_region_created
				}
			}
			situation:the_christian_church = {
				trigger_situation_catalyst = catalyst_the_christian_church_clerical_region_created
			}
		}
	}
	if = {
		limit = { exists = scope:new_ecclesiastical_holder }
		scope:actor = {
			# send_interface_toast = {
				# type = event_toast_effect_good
				# title = request_ecclesiastical_title_creation_accepted_tt
				# left_icon = scope:new_title
				# right_icon = scope:new_ecclesiastical_holder
				# New holder is grateful — skipped when the petitioner appointed themselves (clergy path), to avoid a self-opinion bonus
				if = {
					limit = {
						scope:new_ecclesiastical_holder != scope:puppet_or_actor
					}
					reverse_add_opinion = {
						target = scope:new_ecclesiastical_holder
						modifier = grateful_opinion
						opinion = 50
					}
				}
			# }
			if = {
				limit = {
					NOT = {
						has_government = ecclesiastical_government
					}
				}
				add_hook = {
					type = loyalty_hook
					target = scope:new_ecclesiastical_holder
				}
			}
		}
		scope:new_ecclesiastical_holder = {
			if = {
				limit = {
					is_ruler = yes # Holder becomes a ruler only after resolve_title_and_vassal_change; guard prevents a tooltip crash
					NOT = { government_has_flag = government_is_ecclesiastical }
				}
				change_government = ecclesiastical_government
			}
		}
	}
	scope:new_ecclesiastical_holder = {
		faith = {
			change_fervor = {
				value = notable_fervor_gain
				desc = fervor_gain_clerical_title_created
			}
		}
	}
	# Trigger any dependent expansions
	$CLERICAL_TITLE_STORY$ = {
		if = {
			limit = { has_variable_list = clerical_title_dependent_expansions }
			every_in_list = {
				variable = clerical_title_dependent_expansions
				limit = { this.var:title = { has_variable = clerical_title } }
				save_scope_as = dependent_expansion
				set_variable = suppress_event
				scope:kingdom_holder = {
					BTWK_expand_clerical_title_effect = { CLERICAL_TITLE_STORY = scope:dependent_expansion }
				}
				set_variable = dependent_expansion_just_triggered
			}
		}
	}
	if = {
		limit = { NOT = { $CLERICAL_TITLE_STORY$ = { has_variable = suppress_event } } }
		scope:kingdom_holder = { trigger_event = BTWK_clerical_titles.0001 }
		scope:new_ecclesiastical_holder = { 
			if = {
				limit = { NOT = { this = scope:kingdom_holder } }
				trigger_event = BTWK_clerical_titles.0001
			}
		}
		every_player = {
			limit = {
				NOR = {
					this = scope:kingdom_holder
					this = scope:new_ecclesiastical_holder
				}
			}
			trigger_event = BTWK_clerical_titles.0001
		}
	}
	# Clear the flag indicating that dependent expansions that were just triggered
	$CLERICAL_TITLE_STORY$ = {
		if = {
			limit = { has_variable_list = clerical_title_dependent_expansions }
			every_in_list = {
				variable = clerical_title_dependent_expansions
				if = {
					limit = { has_variable = dependent_expansion_just_triggered }
					remove_variable = dependent_expansion_just_triggered
				}
			}
		}
	}
}

BTWK_expand_clerical_title_effect = {
	save_scope_as = puppet_or_actor
	$CLERICAL_TITLE_STORY$ = { save_scope_as = clerical_title_expansion_story }
	$CLERICAL_TITLE_STORY$.var:region = { save_scope_as = effect_region }
	$CLERICAL_TITLE_STORY$.var:title.var:clerical_title ?= { save_scope_as = clerical_title }
	# Make sure tooltip is functional even if clerical title doesn't exist yet
	if = {
		limit = { exists = scope:clerical_title }
		custom_tooltip = BTWK_expand_clerical_title_effect_expand_archbishopric_tt
	}
	else = {
		$CLERICAL_TITLE_STORY$.var:title ?= { save_scope_as = clerical_title_template }
		custom_tooltip = BTWK_expand_clerical_title_effect_expand_archbishopric_does_not_exist_yet_tt
	}
	$CLERICAL_TITLE_STORY$ = {
		set_variable = clerical_title_expanded
		hidden_effect = {
			# Need a while loop, as only counties *currently* adjacent to the 
			# archbishopric will be added in each pass
			while = {
				limit = {
					any_county_in_region = {
						region = scope:effect_region
						NOT = { exists = clerical_region_title }
						# Prevent maxing out the while loop for isolated islands, until this gets changed by Paradox
						any_neighboring_county = { clerical_region_title = scope:clerical_title }
					}
				}
				every_county_in_region = {
					region = scope:effect_region
					limit = { NOT = { exists = clerical_region_title } }
					scope:clerical_title = { expand_clerical_region = prev }
				}
			}
		}
	}
	hidden_effect = {
		if = {
			limit = {
				has_pam_dlc_trigger = yes
				scope:puppet_or_actor = {
					any_character_situation = {
						situation_type = the_christian_church
						situation_top_has_catalyst = catalyst_the_christian_church_clerical_region_expanded
					}
				}
			}
			situation:the_christian_church = {
				trigger_situation_catalyst = catalyst_the_christian_church_clerical_region_expanded
			}
		}
	}
	save_scope_as = expansion_actor
	scope:clerical_title.holder ?= { save_scope_as = recipient }
	scope:puppet_or_actor = {
		faith = {
			change_fervor = {
				value = notable_fervor_gain
				desc = fervor_gain_clerical_title_expanded
			}
		}
	}
	if = {
		limit = {
			scope:ecclesiastical_holder != scope:expansion_actor
		}
		reverse_add_opinion = {
			target = scope:ecclesiastical_holder
			modifier = grateful_opinion
			opinion = 30
		}
	}
	if = {
		limit = { NOT = { $CLERICAL_TITLE_STORY$ = { has_variable = suppress_event } } }
		trigger_event = BTWK_clerical_titles.0002
		scope:clerical_title.holder ?= {
			if = {
				limit = { this != prev }
				trigger_event = BTWK_clerical_titles.0002
			}
		}
		every_player = {
			limit = {
				NOR = {
					this = prev
					has_title = scope:clerical_title
				}
			}
			trigger_event = BTWK_clerical_titles.0002
		}
	}
}
