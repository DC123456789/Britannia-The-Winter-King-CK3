# --- Founder spawn dispatcher (any scope) ---
# Calls the matching per-heresy founder template based on scope:heresy_faith
# and the resurgence stage. Saves scope:founder at scope:birth_location.
#
# Stage logic (named historical heresies):
#   first outbreak:                spawn the primary historical figure
#   second outbreak (resurgence):  spawn the named secondary historical figure
#   third+ outbreak:               spawn the per-heresy anonymous resurgence
#
# Always-anonymous heresies (Paulician, Iconoclast, Adoptionist, Messalian) always use their single anonymous template.
pam_heresy_dispatch_founder_spawn_effect = {
	if = {
		limit = { scope:heresy_faith = faith:lollard }
		if = {
			limit = { exists = scope:is_resurgence }
			if = {
				limit = { NOT = { exists = global_var:pam_heresy_lollard_secondary_fired } }
				pam_heresy_spawn_lollard_secondary_founder_effect = yes
				set_global_variable = { name = pam_heresy_lollard_secondary_fired value = yes }
			}
			else = {
				pam_heresy_spawn_lollard_resurgence_founder_effect = yes
			}
		}
		else = {
			pam_heresy_spawn_lollard_founder_effect = yes
		}
	}
	else_if = {
		limit = { scope:heresy_faith = faith:hussite }
		if = {
			limit = { exists = scope:is_resurgence }
			if = {
				limit = { NOT = { exists = global_var:pam_heresy_hussite_secondary_fired } }
				pam_heresy_spawn_hussite_secondary_founder_effect = yes
				set_global_variable = { name = pam_heresy_hussite_secondary_fired value = yes }
			}

			}
		}
		else = {
			pam_heresy_spawn_hussite_founder_effect = yes
		}
	}
	else_if = {
		limit = { scope:heresy_faith = faith:joachimite }

			}
			else = {

			}

		else = {

		}
	}
	else_if = {
		limit = { scope:heresy_faith = faith:cathar }
		if = {
			limit = { exists = scope:is_resurgence }
			if = {
				limit = { NOT = { exists = global_var:pam_heresy_cathar_secondary_fired } }
				pam_heresy_spawn_cathar_secondary_founder_effect = yes
				set_global_variable = { name = pam_heresy_cathar_secondary_fired value = yes }
			}
			else = {
				pam_heresy_spawn_cathar_resurgence_founder_effect = yes
			}
		}
		else = {
			pam_heresy_spawn_cathar_founder_effect = yes
		}
	}
	else_if = {
		limit = { scope:heresy_faith = faith:waldensian }
		if = {
			limit = { exists = scope:is_resurgence }
			if = {
				limit = { NOT = { exists = global_var:pam_heresy_waldensian_secondary_fired } }
				pam_heresy_spawn_waldensian_secondary_founder_effect = yes
				set_global_variable = { name = pam_heresy_waldensian_secondary_fired value = yes }
			}
			else = {
				pam_heresy_spawn_waldensian_resurgence_founder_effect = yes
			}
		}
		else = {
			pam_heresy_spawn_waldensian_founder_effect = yes
		}
	}
	else_if = {
		limit = { scope:heresy_faith = faith:bogomilist }
		if = {
			limit = { exists = scope:is_resurgence }
			if = {
				limit = { NOT = { exists = global_var:pam_heresy_bogomilist_secondary_fired } }

			}
			else = {

			}
		}

			pam_heresy_spawn_bogomilist_founder_effect = yes
		}
	}
	else_if = {
		limit = { scope:heresy_faith = faith:paulician }
		pam_heresy_spawn_paulician_founder_effect = yes
	}
	else_if = {
		limit = { scope:heresy_faith = faith:iconoclast }
		pam_heresy_spawn_iconoclast_founder_effect = yes
	}
	else_if = {
		limit = { scope:heresy_faith = faith:adoptionist }
		pam_heresy_spawn_adoptionist_founder_effect = yes
	}
	else_if = {
		limit = { scope:heresy_faith = faith:messalian }

# create_rite_from_type takes a literal rite type key, so dispatch on heresy_faith.
