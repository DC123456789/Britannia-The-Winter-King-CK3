set_up_dynamic_spiritual_hof_title_effect = {
	# Sort scopes & starting processes.
	faith = { save_scope_as = my_faith }
	$CREATOR$ = { save_scope_as = hof_creator }
	# Pope always get priority for Chalcedonianism
	if = {
		limit = {
			scope:hof_creator.faith.main_rite = rite:roman_rite
		}
		title:k_papal_state = { save_scope_as = new_title }
	}
	else = {
		create_dynamic_title = {
			tier = duchy
			name = REL_HEAD_TITLE_NAME
			adj = REL_HEAD_TITLE_ADJ
		}
		create_title_and_vassal_change = {
			type = created
			save_scope_as = change
			add_claim_on_loss = no
		}
	}
	# Pope always get priority for Chalcedonianism
	if = {
		limit = {
			scope:new_title = title:k_papal_state
		}
		title:k_papal_state.holder = { save_scope_as = new_religious_head }
	}
	# Clergy take the title, while lay creators prefer clerical puppets over the realm priest
	else_if = {
		limit = {
			scope:hof_creator = { is_clergy = yes }
		}
		scope:hof_creator = { save_scope_as = new_religious_head }
	}
	else_if = {
		limit = {
			scope:hof_creator.puppet:theological_agent_puppet ?= {
				rite = scope:hof_creator.rite
			}
		}
		scope:hof_creator.puppet:theological_agent_puppet = {
			save_scope_as = new_religious_head
		}
	}
	else_if = {
		limit = {
			scope:hof_creator.puppet:external_theological_agent_puppet ?= {
				rite = scope:hof_creator.rite
			}
		}
		scope:hof_creator.puppet:external_theological_agent_puppet = {
			save_scope_as = new_religious_head
		}
	}
	else_if = {
		limit = {
			scope:hof_creator.cp:councillor_court_chaplain ?= {
				rite = scope:hof_creator.rite
			}
		}
		scope:hof_creator.cp:councillor_court_chaplain = {
			save_scope_as = new_religious_head
		}
	}
	else_if = {
		limit = {
			any_theocratic_vassal = { faith = $CREATOR$.faith }
		}
		ordered_theocratic_vassal = {
			limit = { faith = $CREATOR$.faith }
			order_by = head_of_faith_selection_weight
			save_scope_as = new_religious_head
		}
	}
	else = {
		create_character = {
			location = $CREATOR$.capital_province
			age = { 30 50 }
			gender_female_chance = root_faith_clergy_gender_female_chance
			trait = education_learning_4
			faith = $CREATOR$.faith
			rite = $CREATOR$.rite
			culture = $CREATOR$.culture
			learning = { 14 22 }
			save_scope_as = new_religious_head
		}
	}
	# Try to land the new title.
	# BTWK - Pope doesn't get landed when he is elevated (since he's off in Italy), unless you somehow vassalized him
	if = {
		limit = {
			NOT = {
				scope:new_religious_head = {
					has_title = title:k_papal_state
					NOT = { any_liege_or_above = { this = scope:hof_creator } }
				}
			}
		}
		scope:my_faith = {
			random_holy_site_barony = {
				# Prefer a capital that's under $CREATOR$.
				limit = { county.holder = $CREATOR$ }
				# Even if it's indirectly under $CREATOR$.
				alternative_limit = {
					county.holder = { target_is_liege_or_above = $CREATOR$ }
				}
				# Then go for any that's held by someone of the faith.
				alternative_limit = { county.holder.faith = $CREATOR$.faith }
				# Under *extreme* circumstances, pick any holy site with at least some faithful.
				alternative_limit = { county.faith = $CREATOR$.faith }
				# Ok, just pick any of them. We're out of options and we need _somewhere_.
				alternative_limit = { always = yes }
				save_scope_as = holy_site
			}
		}
	}
	if = {
		limit = { exists = scope:holy_site }
		scope:new_title = { set_capital_county = scope:holy_site.county }
	}
	# Configure final effects.
	if = {
		limit = { NOT = { scope:new_title = title:k_papal_state } }
		hidden_effect = {
			scope:new_title = {
				set_landless_title = yes
				set_destroy_if_invalid_heir = yes
				set_no_automatic_claims = yes
				set_definitive_form = yes
				set_always_follows_primary_heir = yes
				change_title_holder = {
					holder = scope:new_religious_head
					change = scope:change
				}
			}
		}
		resolve_title_and_vassal_change = scope:change
	}
	faith = { set_religious_head_title = scope:new_title }
	if = {
		limit = { NOT = { scope:new_title = title:k_papal_state } }
		scope:new_title = { generate_coa = religious_title }
	}
	# The establishing rite becomes the mainline rite of the faith.
	if = {
		limit = {
			scope:hof_creator = {
				exists = rite
				NOT = { rite = scope:my_faith.main_rite }
				NOT = { scope:new_title = title:k_papal_state }		# If we're elevating the Pope, Roman Rite still remains the mainline!
			}
		}
		scope:hof_creator.rite = { set_main_rite = yes }
	}
	# We enclose the law safeguard in an if statement to prevent errors when this is only _about_ to be fired.
	hidden_effect = {
		if = {
			limit = { exists = scope:new_religious_head }
			scope:new_religious_head = {
				if = {
					limit = {
						NOT = { has_realm_law = same_faith_theocratic_succession_law }
					}
					add_realm_law_skip_effects = same_faith_theocratic_succession_law
				}
			}
		}
	}
}
