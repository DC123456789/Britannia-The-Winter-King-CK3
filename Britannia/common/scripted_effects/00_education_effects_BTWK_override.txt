post_education_bonuses_effect = {
	if = {
		limit = {
			exists = scope:educator
			character_has_commander_trait_scope_does_not = scope:educator
			scope:educator = {
				culture = {
					has_cultural_parameter = guardians_can_transfer_commander_traits
				}
			}
		}
		culture_randomize_commander_trait_effect = yes
	}
	
	# BTWK - inherit Roman Military Training
	if = {
		limit = {
			scope:educator ?= { has_trait = roman_trained }
			has_trait_rank = {
				trait = education_martial
				rank >= 1
			}
		}
		random = {
			chance = {
				value = 50
				if = {
					limit = {
						has_trait_rank = {
							trait = education_martial
							rank >= 3
						}
					}
					add = 50
				}
				else_if = {
					limit = {
						has_trait_rank = {
							trait = education_martial
							rank == 2
						}
					}
					add = 25
				}
			}
			add_trait = roman_trained
		}
	}
}
