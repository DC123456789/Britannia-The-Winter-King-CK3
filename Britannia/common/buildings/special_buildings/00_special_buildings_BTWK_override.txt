holy_site_cathedral_01 = {
	effect_desc = {
		triggered_desc = {
			trigger = { has_dlc_feature = legends_of_the_dead }
			desc = building_funeral_effect_desc
		}
	}

	asset = {
		type = pdxmesh
		name = "cp6_building_special_grand_cathedral_mesh"
		requires_dlc_flag = holy_buildings
	}
	asset = {
		type = pdxmesh
		name = "building_special_cathedral_generic_mesh"
	}

	construction_time = very_slow_construction_time

	type_icon = "special_buildings/cologne_cathedral.dds"
	
	can_construct_potential = {
		building_requirement_tribal = no
	}
	
	can_construct = {
		trigger_if = {
			limit = { barony = { has_variable = BTWK_britannia_high_king_state_church_seat_faith } }
			custom_description = {
				text = BTWK_britannia_high_king_state_church_seat_or_holy_site_building_trigger
				OR = {
					barony = {
						is_holy_site_of = scope:holder.faith
					}
					scope:holder.faith = barony.var:BTWK_britannia_high_king_state_church_seat_faith
				}
			}
		}
		trigger_else = {
			custom_description = {
				text = holy_site_building_trigger
				barony = {
					is_holy_site_of = scope:holder.faith
				}
			}
		}
		building_monument_tribal_trigger = yes
	}
	
	is_enabled = {
		trigger_if = {
			limit = { barony = { has_variable = BTWK_britannia_high_king_state_church_seat_faith } }
			custom_description = {
				text = BTWK_britannia_high_king_state_church_seat_or_holy_site_building_trigger
				OR = {
					barony = {
						is_holy_site_of = scope:holder.faith
					}
					scope:holder.faith = barony.var:BTWK_britannia_high_king_state_church_seat_faith
				}
			}
		}
		trigger_else = {
			custom_description = {
				text = holy_site_building_trigger
				barony = {
					is_holy_site_of = scope:holder.faith
				}
			}
		}
	}
	
	cost_gold = 1000

	character_modifier = {
		monthly_piety = 0.25
		monthly_dynasty_prestige_mult = 0.05
		holy_order_hire_cost_mult = -0.15
		clergy_opinion = 5
		character_capital_county_monthly_control_change_add = 0.05
	}
	
	province_culture_modifier = {
		parameter = civic_religion_cathedral_civic_tax
		monthly_civic_tax = 0.5
	}
	province_modifier = {
		monthly_religious_tax = 1
		trade_good_books_paper_production_add = special_building_books_output
	}
	market_modifier = {
		trade_good_incense_sell_price_mult = special_building_need_sell_price
		trade_good_books_paper_buy_price_mult = special_building_production_buy_price
		trade_good_jewelry_sell_price_mult = special_building_need_sell_price
	}

	on_complete = {
		#Regular Mandala Boon
		mandala_built_piety_duchy_building_t3_effect = yes
	}
	
	ai_value = {
		base = 10
		ai_pious_building_preference_modifier = yes
		modifier = { # Fill all building slots before going for special buildings
			factor = 0
			free_building_slots > 0
		}
		ai_special_district_modifier = { LEVEL = 1 }
	}
	

	trade_good_need = {
		trade_good = trade_good_incense
		amount = special_building_need_amount
		can_be_filled_multiple_times = 5

		modifiers = {
			county_modifier = {
				prosperity_growth_factor = special_building_need_prosperity_high
			}
		}
	}

	trade_good_need = {
		trade_good = trade_good_jewelry
		amount = special_building_need_amount
		can_be_filled_multiple_times = 5
		not_rite_parameter = luxury_forbidden

		modifiers = {
			province_modifier = {
				monthly_religious_tax = special_building_need_tax
			}
		}
	}

	flag = travel_point_of_interest_religious
	flag = holy_building
	flag = religious_studies_location
	market_rite_modifier = {
		parameter = can_go_on_pilgrimage
		trade_good_incense_sell_price_mult = holy_site_pilgrim_sell_price_mult
		trade_good_oil_sell_price_mult = holy_site_pilgrim_sell_price_mult
		trade_good_honey_sell_price_mult = holy_site_pilgrim_sell_price_mult
		trade_good_ceramics_sell_price_mult = holy_site_pilgrim_sell_price_mult
	}
	market_rite_modifier = {
		parameter = mandatory_pilgrimage
		trade_good_incense_sell_price_mult = holy_site_mandatory_pilgrimage_sell_price_mult
		trade_good_oil_sell_price_mult = holy_site_mandatory_pilgrimage_sell_price_mult
		trade_good_honey_sell_price_mult = holy_site_mandatory_pilgrimage_sell_price_mult
		trade_good_ceramics_sell_price_mult = holy_site_mandatory_pilgrimage_sell_price_mult
	}
	market_rite_modifier = {
		parameter = mandatory_hajj
		trade_good_incense_sell_price_mult = holy_site_mandatory_pilgrimage_sell_price_mult
		trade_good_oil_sell_price_mult = holy_site_mandatory_pilgrimage_sell_price_mult
		trade_good_honey_sell_price_mult = holy_site_mandatory_pilgrimage_sell_price_mult
		trade_good_ceramics_sell_price_mult = holy_site_mandatory_pilgrimage_sell_price_mult
	}
}
