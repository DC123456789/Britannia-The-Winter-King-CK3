# Determines if a character can convert faith via the convert faith UI, on top of the piety requirement
# Root is the character that would convert
# scope:new_faith is the faith they want to convert to
faith_conversion = {
	is_adult = yes
	faith.religious_head != root
	custom_tooltip = {
		text = is_antipope_tt
		pam_is_antipope_trigger = no
	}
	is_in_ongoing_great_holy_war = no
	trigger_if = {
		limit = { has_variable = BTWK_rite_to_convert_to }
		this.var:BTWK_rite_to_convert_to = { save_temporary_scope_as = new_rite }
	}
	trigger_else = {
		scope:new_faith.main_rite = { save_temporary_scope_as = new_rite }
	}
	scope:new_rite = {
		is_rite_enabled = yes
		can_convert_to_rite = yes
	}

	# Can't (or shouldn't) convert to faiths that are meant to be resurrected via decision, event, etc.
	custom_tooltip = {
		text = faith_conversion_cost_conversion_blocked_till_decision_taken
		NOT = {
			scope:new_faith = { has_variable = block_conversion_till_decision_taken }
		}
	}
	custom_tooltip = {
		text = faith_conversion_cost_conversion_blocked_till_nebulous_circumstances
		NOT = {
			scope:new_faith = { has_variable = block_conversion_till_nebulous_circumstances }
		}
	}

	# The realm teaches its own State Rite: converting to its faith skips the recency and knowledge gates
	trigger_if = {
		limit = {
			exists = top_liege.primary_title.state_rite
			scope:new_faith = top_liege.primary_title.state_rite.faith
		}
		always = yes
	}
	trigger_else = {
		custom_tooltip = {
			text = faith_conversion_recently_converted
			NOT = {
				has_character_flag = faith_conversion_recently_converted
			}
		}

		OR = {
			AND = {
				faith.religion = scope:new_faith.religion
				"knows_rite_level(scope:new_rite)" > pam_same_religion_rite_knowledge_requirement
			}
			"knows_rite_level(scope:new_rite)" > pam_other_religion_rite_knowledge_requirement
			# If your capital is Christian, you can always convert to it
			trigger_if = {
				limit = {
					scope:new_faith.religion = religion:christianity_religion
				}
				custom_tooltip = {
					text = faith_conversion_BTWK_spreading_of_christianity_faith_capital
					capital_county.faith = scope:new_faith
				}
			}
			# Can always convert to any Christian faith with a significant presence in your region
			# Player only just in case for performance reasons
			trigger_if = {
				limit = {
					is_ai = no
					scope:new_faith.religion = religion:christianity_religion
				}
				save_temporary_scope_as = current_character
				custom_tooltip = {
					text = faith_conversion_BTWK_spreading_of_christianity_faith_spread
					situation:spreading_of_christianity_situation = {
						any_participant_group = {
							participant_group_has_character = scope:current_character
							participant_group_sub_region = {
								any_situation_sub_region_county = {
									percent >= 0.15
									faith = scope:new_faith
								}
							}
						}
					}
				}
			}
		}
	}
	
	# BTWK blocks
	custom_tooltip = {		# Activated by event
		text = faith_conversion_cost_conversion_blocked_till_founded
		NOT = {
			scope:new_faith = { has_variable = block_conversion_till_founded }
		}
	}
	custom_tooltip = {		# Completely blocked
		text = faith_conversion_cost_conversion_blocked
		NOT = {
			scope:new_faith = { has_variable = block_conversion }
		}
	}
}

# Is this character allowed to diverge their culture?
# root: Character
can_diverge_culture = {
	culture = { culture_age >= culture_divergence_cooldown }
	custom_description = {
		text = culture_present_in_realm_for_divergence
		any_sub_realm_county = {
			culture = root.culture
		}
	}
	BTWK_culture_allowed_to_diverge_at_county_trigger = {
		CULTURE = culture
		COUNTY = capital_county
	}
}
