# The liturgical language triggers below only evaluate in a rite scope; call them through
# main_rite from a faith and through rite from a character.
# Pre-scripted rites skip the Head of Rite culture fallback. Slavic, Georgian, Antiochian,
# heresies, and custom divergences keep that fallback.
rite_has_pre_scripted_liturgical_language_trigger = {
	OR = {
		faith = faith:catholic
		this = rite:roman_rite
		this = rite:ambrosian_rite
		this = rite:beneventan_rite
		this = rite:mozarabic_church
		this = rite:carolingian_christianity
		faith = faith:orthodox
		this = rite:byzantine_rite
		this = rite:insular_celtic
		this = rite:ethiopian_rite
		this = rite:armenian_rite
		this = rite:coptic_rite
		# BTWK
		this = rite:britannic_rite
		this = rite:saxon_rite
	}
}

# These are founding-see rites in rite scope, not map-wide or ordinary swap traditions
pam_rite_is_regional_trigger = {
	OR = {
		this = rite:insular_celtic
		this = rite:georgian_rite
		this = rite:ambrosian_rite
		this = rite:beneventan_rite
		this = rite:mozarabic_church
		this = rite:carolingian_christianity
		this = rite:slavic_rite
		this = rite:coptic_rite
		this = rite:antiochian_rite
		this = rite:ethiopian_rite
		this = rite:armenian_rite
		
		# BTWK
		this = rite:britannic_rite
		this = rite:saxon_rite
	}
}

pam_latin_as_liturgical_language_trigger = {
	OR = {
		faith = faith:catholic
		this = rite:roman_rite
		this = rite:ambrosian_rite
		this = rite:beneventan_rite
		this = rite:mozarabic_church
		this = rite:carolingian_christianity
		# BTWK
		this = rite:britannic_rite
		this = rite:saxon_rite
		AND = {
			rite_has_pre_scripted_liturgical_language_trigger = no
			head_of_rite.culture ?= {
				OR = {
					this = culture:italian
					has_same_culture_language = culture:italian
				}
			}
		}
	}
}

# Great Schism: progress can still change. Any scope
pam_great_schism_progress_can_change_trigger = {
	always = no		# Frozen in pre-schism state in BTWK
	NOT = { has_global_variable = pam_great_schism_decision_taken }
	exists = situation:the_christian_church
}

is_clerical_appointment_valid_trigger = {
	$TITLE$ = { save_temporary_scope_as = title_temp }
	$CANDIDATE$ = { save_temporary_scope_as = candidate_temp }
	$TOP_LIEGE$ = { save_temporary_scope_as = top_liege_temp }
	trigger_if = {
		limit = { exists = scope:title_temp }
		custom_tooltip = { # Has relevant law
			text = has_appointment_succession_tt
			$TITLE$ = {
				OR = {
					has_title_law = clerical_appointment_law
					holder ?= { has_realm_law = clerical_appointment_law }
				}
			}
		}
		custom_tooltip = { # Titles below the holder's primary tier are inherited alongside it, so they cannot be contested on their own
			text = follows_other_title_appointment_tt
			scope:title_temp.holder ?= { primary_title.tier = scope:title_temp.tier }
		}
	}
	is_clerical_appointment_valid_base_trigger = {
		CANDIDATE = $CANDIDATE$
		TOP_LIEGE = $TOP_LIEGE$
	}
	trigger_if = {		# BTWK
		limit = { scope:title_temp = title:k_papal_state }
		scope:candidate_temp = { 
			custom_tooltip = {
				text = BTWK_can_be_elected_pope_trigger_tt
				BTWK_can_be_elected_pope_trigger = yes 
			}
		}
	}
}

pam_beseech_claims_access_trigger = {

	}
}

actor_incite_holy_war_shown_trigger = {
	always = no		# Disabled in BTWK
	is_ruler = yes
	OR = {
		government_has_flag = government_is_ecclesiastical
		this = religious_head_or_challenger
	}
	faith = {
		NOR = {
			has_doctrine_parameter = holy_wars_forbidden
			has_doctrine_parameter = unreformed
		}
	}
}
