tribal_subjugation_cb_prestige_cost = {
	value = 0
	if = {
		limit = { BTWK_britannia_is_high_king_subjugation_trigger = yes }
		add = {
			value = 500
			desc = CB_BASE_COST
		}
	}
	else = {
		if = {
			limit = {
				NAND = {
					scope:attacker.faith.religious_head = scope:attacker
					scope:defender.faith = scope:attacker.faith
				}
			}
			add = {
				value = 1000
				desc = CB_BASE_COST
			}
		}
		if = {
			limit = {
				exists = scope:attacker.primary_title
			}
			if = {
				limit = {
					scope:attacker.primary_title = {
						tier = tier_kingdom
					}
				}
				multiply = {
					value = 1.5
					desc = CB_COST_RULES_KINGDOM
				}
			}
			else_if = {
				limit = {
					scope:attacker.primary_title = {
						tier = tier_empire
					}
				}
				multiply = {
					value = 2.5
					desc = CB_COST_RULES_EMPIRE
				}
			}
			else_if = {
				limit = {
					scope:attacker.primary_title = {
						tier = tier_hegemony
					}
				}
				multiply = {
					value = 3.5
					desc = CB_COST_RULES_HEGEMONY
				}
			}
		}
	}

	multiply = common_cb_prestige_cost_multiplier
}

county_cb_prestige_base_cost = {
	add = {
		value = 50
		desc = CB_BASE_COST
	}
	if = {
		limit = {
			scope:defender.faith != scope:attacker.faith
		}
		add = {
			value = 50
			desc = CB_COST_DIFFERENT_FAITH_NOT_BASELINE
		}
		add = {
			value = 5
			multiply = scope:attacker.sub_realm_size
			desc = CB_REALM_SIZE_COST
			max = 100
		}
	}
}
