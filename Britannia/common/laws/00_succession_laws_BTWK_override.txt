
male_only_law = {
	law_group_type = succession_gender_laws
	index = 0

	can_title_have = {
		can_title_have_law_general_trigger = yes
	}
	should_show_for_title = { # Never show this in the UI, just apply it through script
		always = no
	}
	can_keep = {
		trigger_if = {
			limit = {
				can_have_male_monastic_holy_order_succession_law_trigger = yes
			}
			always = yes
		}
		trigger_else_if = {
			limit = {
				government_has_flag = government_is_theocracy
			}
			rite = { rite_has_parameter = clergy_must_be_male }
		}
		trigger_else_if = {
			limit = { should_synchronize_liege_law = yes }
			top_liege = { has_realm_law = male_only_law }
		}
		trigger_else = {
			custom_tooltip = {
				text = culture_not_male_only_tt
				rite = { rite_has_parameter = male_dominated_law }
				NOT = {
					culture = { has_cultural_parameter = female_only_inheritance }
				}
			}
		}
	}
	can_have = {
		trigger_if = {
			limit = {
				government_has_flag = government_is_theocracy
				NOT = { can_have_male_monastic_holy_order_succession_law_trigger = yes }
			}
			rite = { rite_has_parameter = clergy_must_be_male }
		}
	}
	can_pass = {
		can_change_succession_law_trigger = yes
	}
	should_start_with = {
		trigger_if = {
			limit = {
				can_have_male_monastic_holy_order_succession_law_trigger = yes
			}
			always = yes
		}
		trigger_else_if = {
			limit = { should_synchronize_liege_law = yes }
			top_liege = { has_realm_law = male_only_law }
		}
		trigger_else_if = {
			limit = {
				government_has_flag = government_has_merit
			}
			rite = {
				rite_has_parameter = male_dominated_law
			}
		}
		trigger_else_if = {
			limit = {
				government_has_flag = government_is_theocracy
			}
			rite = {
				rite_has_parameter = clergy_must_be_male
			}
		}
		trigger_else = {
			# AND = {
				# faith = {
					# religion_tag = islam_religion
				# }
				# rite = {
					# rite_has_parameter = male_dominated_law
				# }
			# }
			always = yes
		}
	}
	succession = {
		gender_law = male_only
	}
	flag = men_can_have_multiple_spouses
	flag = men_can_have_consorts
	flag = no_female_commanders

	pass_cost = {
		prestige = change_succession_law_prestige_cost
	}
	revoke_cost = {
		prestige = change_succession_law_prestige_cost
	}
}

male_preference_law = {
	law_group_type = succession_gender_laws
	index = 1

	can_title_have = {
		can_title_have_law_general_trigger = yes
	}
	should_show_for_title = { # Never show this in the UI, just apply it through script
		always = no
	}
	can_keep = {
		trigger_if = {
			limit = {
				government_has_flag = government_is_theocracy
			}
			custom_tooltip = {
				text = gender_preference_law_not_for_clergy_tt
				always = no
			}
		}
		trigger_else_if = {
			limit = { should_synchronize_liege_law = yes }
			top_liege = { has_realm_law = male_preference_law }
		}
		trigger_else = {
			custom_description = {
				text = change_law_male_preference_faith_requirement
				rite = {
					NOT = { rite_has_parameter = female_dominated_law }
				}
			}
			custom_tooltip = {
				text = male_preference_law_can_keep_faith_or_culture_trigger
				OR = {
					rite = { rite_has_parameter = male_dominated_law }
					NOT = {
						culture = { has_cultural_parameter = female_only_inheritance }
					}
				}
			}
		}
	}
	can_have = {
		trigger_if = {
			limit = {
				government_has_flag = government_is_theocracy
			}
			custom_tooltip = {
				text = gender_preference_law_not_for_clergy_tt
				always = no
			}
		}
	}
	can_pass = {
		can_change_succession_law_trigger = yes
	}
	should_start_with = {
		trigger_if = {
			limit = { should_synchronize_liege_law = yes }
			top_liege = { has_realm_law = male_preference_law }
		}
		trigger_else = {
			NOT = {
				culture = { has_cultural_parameter = female_preference_inheritance }
			}
			OR = {
				faith = {
					# NOR = {
						# religion_tag = islam_religion
						# AND = {
							# root = {
								# government_has_flag = government_has_merit
							# }
							# root.rite = {
								# rite_has_parameter = male_dominated_law
							# }
						# }
					# }
					always = no
					root.rite = {
						rite_has_parameter = male_dominated_law
					}
				}
				AND = {
					has_realm_law = male_only_law # For when male_only_law ceases to be a legal option
					trigger_if = {
						limit = { should_synchronize_liege_law = yes }
						NOT = {
							top_liege = { has_realm_law = male_only_law }
						}
					}
					trigger_else = {
						custom_tooltip = {
							text = culture_not_female_only_tt
							NOR = {
								rite = { rite_has_parameter = male_dominated_law }
								NOT = {
									culture = { has_cultural_parameter = female_only_inheritance }
								}
							}
						}
					}
				}
			}
		}
	}
	succession = {
		gender_law = male_preference
	}
	flag = men_can_have_multiple_spouses
	flag = men_can_have_consorts
	flag = no_female_commanders

	pass_cost = {
		prestige = change_succession_law_prestige_cost
	}
	revoke_cost = {
		prestige = change_succession_law_prestige_cost
	}
}
