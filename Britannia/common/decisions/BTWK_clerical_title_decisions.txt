BTWK_create_clerical_titles_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/BTWK_spreading_of_christianity.dds"
	}
	decision_group_type = realm

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 0
		kingdom = 36
		empire = 36
		hegemony = 36
	}

	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					has_variable = selected_title
					this.var:selected_title ?= {
						OR = {
							this.var:title = title:d_et_london
							this.var:title = title:d_et_lincoln
							this.var:title = title:d_et_cirencester
							this.var:title = title:d_et_york
						}
					}
					NOT = { has_global_variable = end_of_britannia }
				}
				desc = BTWK_create_clerical_titles_decision_roman_britain_sees_desc
			}
			desc = BTWK_create_clerical_titles_decision_desc
		}
	}		

	is_shown = {
		religion = religion:christianity_religion
		trigger_if = {
			limit = { is_ai = yes }
			is_independent_ruler = yes
		}
		any_in_global_list = {
			variable = creatable_clerical_title_stories
			ROOT = {
				BTWK_potential_create_clerical_title_trigger = {
					CLERICAL_TITLE_STORY = prev
				}
			}
		}
	}

	is_valid_showing_failures_only = {
		BTWK_can_create_clerical_title_common_trigger = yes
		custom_tooltip = {
			text = BTWK_create_clerical_titles_decision_any_formable_tt
			any_in_global_list = {
				variable = creatable_clerical_title_stories
				ROOT = { BTWK_can_create_clerical_title_trigger = { CLERICAL_TITLE_STORY = prev } }
			}
		}
	}

	effect = {
		# This part is purely for display purposes
		if = {
			limit = {
				is_ai = no
				has_variable = selected_title
			}
			show_as_tooltip = {
				ROOT = { 
					BTWK_create_clerical_title_cost_effect = yes
					BTWK_create_clerical_title_effect = { CLERICAL_TITLE_STORY = var:selected_title }
				}
			}
		}
		hidden_effect = {
			while = {
				limit = {
					any_in_global_list = {
						variable = creatable_clerical_title_stories
						ROOT = { 
							BTWK_can_create_clerical_title_trigger = { CLERICAL_TITLE_STORY = prev } 
							BTWK_can_create_clerical_title_cost_trigger = yes
						}
					}
				}
				random_in_global_list = {
					variable = creatable_clerical_title_stories
					limit = {
						ROOT = { 
							BTWK_can_create_clerical_title_trigger = { CLERICAL_TITLE_STORY = prev } 
							BTWK_can_create_clerical_title_cost_trigger = yes
						}
					}
					ROOT = {
						BTWK_create_clerical_title_cost_effect = yes
						BTWK_create_clerical_title_effect = {
							CLERICAL_TITLE_STORY = prev
						}
					}
				}
			}
			remove_variable = selected_title
		}
	}
	
	ai_potential = {
		is_independent_ruler = yes
	}

	ai_will_do = {
		base = 100
	}
	
	widget = {
		gui = "decision_view_widget_create_clerical_titles"
		controller = decision_option_list_controller

		### This placeholder has to be here or the game will crash
		item = {
			value = none
			is_shown = { always = yes }
			is_valid = { always = yes }
			localization = placeholder_name
		}
	}
}
