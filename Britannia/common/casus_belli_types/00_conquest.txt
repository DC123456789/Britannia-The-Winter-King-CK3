county_conquest_cb = {
	icon = county_conquest_cb
	group = conquest

	combine_into_one = yes
	should_show_war_goal_subview = yes
	mutually_exclusive_titles = { always = yes }

	allowed_for_character = {
		can_use_conquest_cbs_trigger = yes
		NOT = {
			government_has_flag = government_is_landless_adventurer
		}
	}

	allowed_for_character_display_regardless = {
		# Struggle constraint - Note: This is the CB you get in FP2 opportunity, if you need to debug, test it here first.
		custom_tooltip = {
			text = struggle_involvement_cb_check_failure
			struggle_blocks_invasion_conquest_cb_trigger = yes
		}
		tgp_japan_offensive_wars_ban_trigger = yes
		trigger_if = {
			limit = {
				government_has_flag = government_has_tribal_fame
			}
			prestige_level >= 2
		}
	}

	allowed_against_character = {
		scope:attacker.top_liege != scope:defender.top_liege
		NAND = {
			scope:attacker = { government_has_flag = government_is_nomadic }
			scope:defender = { government_has_flag = government_is_nomadic }
		}
		scope:attacker = {
			# If you are Tribal or Folkright...
			trigger_if = {
				limit = {
					government_has_flag = government_limited_conquest_cb
				}
				OR = {
					# You are an unromanized Germanic who is not a foederati of someone who is not an unromanized Germanic,
					# and you are attacking someone who is not unromanized Germanic
					# Or you are not tribal (Folkright unromanized Germanic can declare on anyone)
					AND = {
						culture = { has_cultural_parameter = unromanized_germanic_access_to_conquest_cbs }
						NOT = {
							AND = {
								is_tributary = yes
								has_subject_contract_group = tributary_foederatus
								suzerain ?= { NOT = { culture = { has_cultural_parameter = unromanized_germanic_access_to_conquest_cbs } } }
							}
						}
						OR = {
							scope:defender.top_liege.culture = { NOT = { has_cultural_parameter = unromanized_germanic_access_to_conquest_cbs } }
							NOT = { government_has_flag = government_is_tribal }
						}
					}
					# Or you are not an unromanized Germanic, and you are attacking an unromanized Germanic 
					# who is not a foederati of someone who is not an unromanized Germanic
					AND = {
						NOT = { culture = { has_cultural_parameter = unromanized_germanic_access_to_conquest_cbs } }
						scope:defender.top_liege = {
							culture = { has_cultural_parameter = unromanized_germanic_access_to_conquest_cbs }
							NOT = {
								AND = {
									is_tributary = yes
									has_subject_contract_group = tributary_foederatus
									suzerain ?= { NOT = { culture = { has_cultural_parameter = unromanized_germanic_access_to_conquest_cbs } } }
								}
							}
						}
					}
					# Or you have the Adventurer Warlord modifier, and you are attacking someone outside your heritage
					AND = {
						has_character_modifier = BTWK_recent_adventurer_conquest_modifier
						NOT = { has_same_culture_heritage = scope:defender.top_liege.culture }
					}
					# Or you are attacking someone outside your heritage who has the Adventurer Warlord modifier, and 
					AND = {
						scope:defender = { has_character_modifier = BTWK_recent_adventurer_conquest_modifier }
						NOT = { has_same_culture_heritage = scope:defender.top_liege.culture }
					}
				}
			}
		}
	}
	target_titles = neighbor_land
	target_title_tier = county
	ignore_effect = change_title_holder

	attacker_score_from_occupation_scale = 150
	attacker_ticking_warscore = 0.1
	defender_score_from_occupation_scale = 75
	attacker_score_from_battles_scale = 150
	defender_score_from_battles_scale = 150

	ai_only_against_neighbors = yes
	# ai_can_target_all_titles = {
		# can_use_viking_invasion_cbs_trigger = yes
	# }
	ai_score_mult = {
		value = viking_conquest_ai_score_value

		add = common_cb_ai_score_mult_additive
		multiply = aggressive_cb_ai_score_mult
		multiply = common_cb_ai_score_mult
	}

	valid_to_start = {
		scope:target = {
			is_landless_type_title = no
			neighboring_county_or_viking_conquest_trigger = { CHARACTER = root }
		}
		
		# AI Aelle directives - block needed since he is a conqueror
		NOT = {
			AND = {
				scope:attacker = {
					OR = {
						has_character_flag = aelle_of_anglia
						top_liege = { has_character_flag = aelle_of_anglia }
					}
					is_ai = yes
				}
				OR = {
					# Stop AI Aelle from going to the north
					scope:target = {
						title_province = {
							OR = {
								geographical_region = btwk_custom_lindum
								geographical_region = btwk_world_britannia_hen_ogledd
							}
						}
					}
					# Make sure Aelle consolidates the border region before going further west
					AND = {
						scope:attacker = { 
							has_character_flag = aelle_of_anglia
							NOT = {
								any_county_in_region = {
									region = btwk_custom_aelle_target_region
									percent >= 0.85
									holder.top_liege = {
										OR = {
											this = scope:attacker
											suzerain ?= scope:attacker
										}
									}
								}
							}
						}
						scope:target = {
							title_province = {
								NOT = { geographical_region = btwk_custom_aelle_target_region }
							}
						}
					}
				}
			}
		}
		
		# BTWK AI cultural redoubt blocks
		scope:target = { BTWK_ai_cb_target_title_block_on_cultural_redoubts_trigger = yes }
		
		trigger_if = {
			limit = { 
				scope:attacker = {
					culture = { has_cultural_parameter = unromanized_germanic_access_to_conquest_cbs } 
					any_character_struggle = {
						involvement = involved
						has_struggle_phase_parameter = unromanized_germanic_conquest_cb_on_laeti_blocked
					}
				} 
			}
			scope:target = {
				custom_tooltip = {
					text = BTWK_cannot_attack_laeti_during_struggle_phase_tt
					NOT = {
						holder = {
							culture = { has_cultural_parameter = unromanized_germanic_access_to_conquest_cbs }
							vassal_contract_has_flag = has_laetus_contract
						}
					}
				}
			}
		}
	}

	should_invalidate = {
		war_should_invalidate_defender_lost_target_lands_trigger = yes
	}

	on_invalidated_desc = msg_county_conquest_war_invalidated_message
	on_invalidated = {}

	cost = {
		piety = {
			add = county_cb_piety_base_cost
			multiply = common_cb_piety_cost_multiplier
		}
		prestige = {
			add = county_cb_prestige_base_cost
			if = {
				limit = {
					scope:attacker = { government_has_flag = government_limited_conquest_cb }
				}
				if = {
					limit = { scope:attacker = { government_has_flag = government_is_tribal } }
					add = {
						value = 200		# Adjustment for 0.5x from Tribal
						desc = CB_COST_LIMITED_CONQUEST_CBS
					}
				}
				else = {
					add = {
						value = 100
						desc = CB_COST_LIMITED_CONQUEST_CBS
					}
				}				
			}
			# Unromanized Germanic attacking another Unromanized Germanic is more expensive
			if = {
				limit = {
					scope:attacker = { 
						culture = { has_cultural_parameter = unromanized_germanic_access_to_conquest_cbs }
						government_has_flag = government_limited_conquest_cb
					}
					scope:defender.top_liege.culture = { has_cultural_parameter = unromanized_germanic_access_to_conquest_cbs } 
				}
				# Especially in the early game
				if = {
					limit = {
						scope:attacker = {
							any_character_struggle = {
								involvement = involved
								has_struggle_phase_parameter = more_expensive_unromanized_germanic_mutual_conquest_cb
							}
						}
						scope:defender = {
							OR = {
								any_character_struggle = {
									involvement = involved
									has_struggle_phase_parameter = more_expensive_unromanized_germanic_mutual_conquest_cb
								}
								any_character_struggle = {
									involvement = interloper
									has_struggle_phase_parameter = more_expensive_unromanized_germanic_mutual_conquest_cb
								}
							}
						}
					}
					add = {
						value = 500
						desc = CB_COST_OTHER_UNROMANIZED_GERMANIC
					}
				}
				else = {
					add = {
						value = 200
						desc = CB_COST_OTHER_UNROMANIZED_GERMANIC
					}
				}
			}
			# Struggle modifiers
			if = {
				limit = {
					scope:attacker = {
						any_character_struggle = {
							involvement = involved
							has_struggle_phase_parameter = more_expensive_conquest_cb
							save_temporary_scope_as = current_struggle
							scope:defender = {
								OR = {
									any_character_struggle = {
										involvement = involved
										this = scope:current_struggle
									}
									any_character_struggle = {
										involvement = interloper
										this = scope:current_struggle
									}
								}
							}
						}
					}
				}
				multiply = {
					value = 2
					desc = CB_STRUGGLE_PHASE_SURCHARGE
				}
			}
			multiply = common_cb_prestige_cost_multiplier
		}
	}

	on_declaration = {
		on_declared_war = yes
	}

	on_victory_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:attacker = { is_local_player = yes } }
				desc = county_conquest_cb_victory_desc_attacker
			}
			desc = county_conquest_cb_victory_desc
		}
		triggered_desc = {			# Warning that the cb is temporary (if you are adventurer)
			trigger = {
				scope:attacker = {
					government_has_flag = government_limited_conquest_cb
					NOR = {
						# You are an unromanized Germanic, and you are attacking someone who is not unromanized Germanic
						# Or you are not tribal (Folkright unromanized Germanic can declare on anyone)
						AND = {
							culture = { has_cultural_parameter = unromanized_germanic_access_to_conquest_cbs }
							OR = {
								scope:defender.top_liege.culture = { NOT = { has_cultural_parameter = unromanized_germanic_access_to_conquest_cbs } }
								NOT = { government_has_flag = government_is_tribal }
							}
						}
						# Or you are not an unromanized Germanic, and you are attacking an unromanized Germanic 
						AND = {
							NOT = { culture = { has_cultural_parameter = unromanized_germanic_access_to_conquest_cbs } }
							scope:defender.top_liege.culture = { has_cultural_parameter = unromanized_germanic_access_to_conquest_cbs }
						}
					}
					# You have the Adventurer Warlord modifier, and you are attacking someone outside your heritage
					AND = {
						has_character_modifier = BTWK_recent_adventurer_conquest_modifier
						NOT = { has_same_culture_heritage = scope:defender.top_liege.culture }
					}
				}
			}
			desc = county_conquest_cb_recent_adventurer_attacker_war_warning_desc
		}
		triggered_desc = {			# Warning that the cb is temporary (if you are attacking adventurer)
			trigger = {
				scope:attacker = {
					government_has_flag = government_limited_conquest_cb
					NOR = {
						# You are an unromanized Germanic, and you are attacking someone who is not unromanized Germanic
						# Or you are not tribal (Folkright unromanized Germanic can declare on anyone)
						AND = {
							culture = { has_cultural_parameter = unromanized_germanic_access_to_conquest_cbs }
							OR = {
								scope:defender.top_liege.culture = { NOT = { has_cultural_parameter = unromanized_germanic_access_to_conquest_cbs } }
								NOT = { government_has_flag = government_is_tribal }
							}
						}
						# Or you are not an unromanized Germanic, and you are attacking an unromanized Germanic 
						AND = {
							NOT = { culture = { has_cultural_parameter = unromanized_germanic_access_to_conquest_cbs } }
							scope:defender.top_liege.culture = { has_cultural_parameter = unromanized_germanic_access_to_conquest_cbs }
						}
					}
					# You are attacking someone outside your heritage who has the Adventurer Warlord modifier 
					AND = {
						scope:defender = { has_character_modifier = BTWK_recent_adventurer_conquest_modifier }
						NOT = { has_same_culture_heritage = scope:defender.top_liege.culture }
					}
				}
			}
			desc = county_conquest_cb_recent_adventurer_defender_war_warning_desc
		}
		triggered_desc = {			# Warning for Cultural Redoubts
			trigger = { 
				BTWK_cb_target_titles_has_cultural_redoubt_trigger = yes
			}
			desc = conquest_cb_cultural_redoubt_warning
		}
	}

	on_victory = {
		on_victory_aggressive_war_effect = yes

		#EP2 accolade glory gain for winning against higher ranked enemy
		scope:attacker = { accolade_attacker_war_end_glory_gain_low_effect = yes }

		create_title_and_vassal_change = {
			type = conquest
			save_scope_as = change
			add_claim_on_loss = yes
		}

		#As this is just a county conquest, simply grab the designated county.
		every_in_list = {
			list = target_titles
			change_title_holder = {
				holder = scope:attacker
				change = scope:change
			}
		}

		resolve_title_and_vassal_change = scope:change

		# Prestige Progress for the Attacker
		every_in_list = {
			list = target_titles
			scope:attacker = {
				add_prestige_experience = minor_prestige_value
			}
		}

		# Allies on both sides get full prestige value for helping in the war, based on their war participation.
		modify_allies_of_participants_fame_values = {
			WINNER = scope:attacker
			FAME_BASE = minor_prestige_value
			IS_RELIGIOUS_WAR = no
			WINNER_ALLY_FAME_SCALE = 1
			LOSER_ALLY_FAME_SCALE = 1
		}
	}

	on_white_peace_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:defender = { is_local_player = yes } }
				desc = county_conquest_cb_white_peace_desc_defender
			}
			desc = county_conquest_cb_white_peace_desc
		}
	}

	on_white_peace = {
		on_white_peace_effect = yes
		# Prestige loss for the attacker
		scope:attacker = {
			add_prestige = {
				value = minor_prestige_value
				multiply = -1.0
			}
		}

		# Allies on both sides get full prestige value for helping in the war, based on their war participation.
		modify_allies_of_participants_fame_values = {
			WINNER = scope:attacker # Not important as the scales are identical
			FAME_BASE = minor_prestige_value
			IS_RELIGIOUS_WAR = no
			WINNER_ALLY_FAME_SCALE = 1
			LOSER_ALLY_FAME_SCALE = 1
		}
	}

	on_defeat_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:defender = { is_local_player = yes } }
				desc = county_conquest_cb_defeat_desc_defender
			}
			triggered_desc = {
				trigger = {
					scope:attacker = {
						is_local_player = yes
						has_targeting_faction = yes
					}
				}
				desc = county_conquest_cb_defeat_desc_attacker
			}
			desc = county_conquest_cb_defeat_desc
		}
	}

	on_defeat = {
		on_defeat_aggressive_war_effect = yes

		scope:defender = {
			#EP2 accolade glory gain for winning against higher ranked enemy
			accolade_defender_war_end_glory_gain_low_effect = yes
			# Prestige for Defender
			add_prestige_war_defender_effect = {
				PRESTIGE_VALUE = medium_prestige_value
			}
		}

		# Prestige loss for the attacker
		scope:attacker = {
			pay_short_term_gold_reparations_effect = {
				GOLD_VALUE = 2
			}
			add_prestige = medium_prestige_loss
		}

		# Allies on both sides get full prestige value for helping in the war, based on their war participation.
		modify_allies_of_participants_fame_values = {
			WINNER = scope:defender
			FAME_BASE = minor_prestige_value
			IS_RELIGIOUS_WAR = no
			WINNER_ALLY_FAME_SCALE = 1
			LOSER_ALLY_FAME_SCALE = 1
		}
	}

	transfer_behavior = transfer

	on_primary_attacker_death = inherit
	on_primary_defender_death = inherit

	attacker_allies_inherit = yes
	defender_allies_inherit = yes

	war_name = "CONQUEST_WAR_NAME"
	war_name_base = "CONQUEST_WAR_NAME_BASE"
	cb_name = "CONQUEST_COUNTY_CB_NAME"

	interface_priority = 79

	use_de_jure_wargoal_only = yes

	attacker_wargoal_percentage = 0.8

	max_defender_score_from_occupation = 150
	max_attacker_score_from_occupation = 150
}

ireland_laudabiliter_conquest_cb = {
	icon = county_conquest_cb
	group = conquest

	combine_into_one = yes
	should_show_war_goal_subview = yes
	mutually_exclusive_titles = { always = yes }

	attacker_score_from_occupation_scale = 150
	attacker_score_from_battles_scale = 150
	defender_score_from_battles_scale = 150

	allowed_for_character = {
		exists = scope:attacker.house
		scope:attacker.house = { #Laudabiliter modifier allows conquest on Ireland region.
			has_house_modifier = ireland_laudabiliter_modifier
		}
	}

	allowed_against_character = {
		scope:attacker.top_liege != scope:defender.top_liege
	}
	target_titles = all
	target_title_tier = county
	target_de_jure_regions_above = yes
	ignore_effect = change_title_holder
	ai_only_against_neighbors = yes

	valid_to_start = {
		scope:target = {
			is_landless_type_title = no
			scope:attacker.house = {
				has_house_modifier = ireland_laudabiliter_modifier
			}
			title_province = { geographical_region = custom_roman_hibernia }
		}
	}

	allowed_for_character_display_regardless = {
		# Struggle constraint
		custom_tooltip = {
			text = struggle_involvement_cb_check_failure
			struggle_blocks_invasion_conquest_cb_trigger = yes
		}
		tgp_japan_offensive_wars_ban_trigger = yes
	}

	should_invalidate = {
		war_should_invalidate_defender_lost_target_lands_trigger = yes
	}

	on_invalidated = {
	}

	cost = {
		piety = {
			add = common_cb_impious_piety_cost
			add = county_cb_piety_base_cost
			multiply = common_cb_piety_cost_multiplier
		}
		prestige = {
			value = 0
			add = county_cb_prestige_base_cost
			multiply = common_cb_prestige_cost_multiplier
		}
	}

	on_declaration = {
		on_declared_war = yes
	}

	on_victory_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:attacker = { is_local_player = yes } }
				desc = ireland_laudabiliter_conquest_cb_victory_desc_attacker
			}
			triggered_desc = {
				trigger = { scope:defender = { is_local_player = yes } }
				desc = ireland_laudabiliter_conquest_cb_victory_desc_defender
			}
			desc = ireland_laudabiliter_conquest_cb_victory_desc
		}
	}

	on_victory = {
		on_victory_aggressive_war_effect = yes

		#EP2 accolade glory gain for winning against higher ranked enemy
		scope:attacker = { accolade_attacker_war_end_glory_gain_low_effect = yes }

		create_title_and_vassal_change = {
			type = conquest
			save_scope_as = change
			add_claim_on_loss = yes
		}

		# go through the dejure hierarchy under target titles, transfer eligible vassals and seize counties from ineligible ones
		every_in_list = {
			list = target_titles
			custom_tooltip = CONQUEST_CB_TITLE

			conquest_cb_title_transfer = {
				RELIGIOUS_WAR = yes
			}
		}

		every_in_list = {
			list = titles_taken
			change_title_holder = {
				holder = scope:attacker
				change = scope:change
				take_baronies = yes
			}
		}

		every_in_list = {
			list = vassals_taken
			change_liege = {
				liege = scope:attacker
				change = scope:change
			}
		}

		resolve_title_and_vassal_change = scope:change

		# Prestige Progress for the Attacker
		every_in_list = {
			list = target_titles
			scope:attacker = {
				add_prestige_experience = minor_prestige_value
			}
		}

		# Allies on both sides get full prestige value for helping in the war, based on their war participation.
		modify_allies_of_participants_fame_values = {
			WINNER = scope:attacker
			FAME_BASE = minor_prestige_value
			IS_RELIGIOUS_WAR = no
			WINNER_ALLY_FAME_SCALE = 1
			LOSER_ALLY_FAME_SCALE = 1
		}
	}

	on_white_peace_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:defender = { is_local_player = yes } }
				desc = ireland_laudabiliter_conquest_cb_white_peace_desc_defender
			}
			desc = ireland_laudabiliter_conquest_cb_white_peace_desc
		}
	}

	on_white_peace = {
		on_white_peace_effect = yes
		# Prestige loss for the attacker
		scope:attacker = {
			add_prestige = {
				value = minor_prestige_value
				multiply = -1.0
			}
		}

		# Allies on both sides get full prestige value for helping in the war, based on their war participation.
		modify_allies_of_participants_fame_values = {
			WINNER = scope:attacker # not important as the scales are identical
			FAME_BASE = minor_prestige_value
			IS_RELIGIOUS_WAR = no
			WINNER_ALLY_FAME_SCALE = 1
			LOSER_ALLY_FAME_SCALE = 1
		}
	}

	on_defeat_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:defender = { is_local_player = yes } }
				desc = ireland_laudabiliter_conquest_cb_defeat_desc_defender
			}
			triggered_desc = {
				trigger = {
					scope:attacker = {
						is_local_player = yes
						has_targeting_faction = yes
					}
				}
				desc = ireland_laudabiliter_conquest_cb_defeat_desc_attacker
			}
			desc = ireland_laudabiliter_conquest_cb_defeat_desc
		}
	}

	on_defeat = {
		on_defeat_aggressive_war_effect = yes

		scope:defender = {
			#EP2 accolade glory gain for winning against higher ranked enemy
			accolade_defender_war_end_glory_gain_low_effect = yes
			# Prestige for Defender
			add_prestige_war_defender_effect = {
				PRESTIGE_VALUE = medium_prestige_value
			}
		}

		# Prestige loss for the attacker
		scope:attacker = {
			pay_short_term_gold_reparations_effect = {
				GOLD_VALUE = 2
			}
			add_prestige = {
				value = medium_prestige_value
				multiply = -1.0
			}
		}

		# Allies on both sides get full prestige value for helping in the war, based on their war participation.
		modify_allies_of_participants_fame_values = {
			WINNER = scope:defender
			FAME_BASE = minor_prestige_value
			IS_RELIGIOUS_WAR = no
			WINNER_ALLY_FAME_SCALE = 1
			LOSER_ALLY_FAME_SCALE = 1
		}
	}

	on_invalidated_desc = msg_county_conquest_war_invalidated_message

	transfer_behavior = transfer

	on_primary_attacker_death = inherit
	on_primary_defender_death = inherit

	attacker_allies_inherit = yes
	defender_allies_inherit = yes

	war_name = "LAUDABILITER_CONQUEST_WAR_NAME"
	war_name_base = "LAUDABILITER_CONQUEST_WAR_NAME_BASE"
	cb_name = "LAUDABILITER_CONQUEST_CB_NAME"

	interface_priority = 80

	use_de_jure_wargoal_only = yes

	attacker_wargoal_percentage = 0.8

	max_defender_score_from_occupation = 150
	max_attacker_score_from_occupation = 150

	max_ai_diplo_distance_to_title = 500

	ai_score_mult = {
		value = 1

		add = common_cb_ai_score_mult_additive
		multiply = aggressive_cb_ai_score_mult
		multiply = common_cb_ai_score_mult
	}
}
