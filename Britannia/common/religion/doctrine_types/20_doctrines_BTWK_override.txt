doctrine_pluralism_fundamentalist = {
	doctrine_group_type = doctrine_pluralism
	index = 0
	divergence = 0

	icon = "doctrine_pluralism_fundamentalist"

	piety_cost = {
		value = faith_doctrine_cost_mid
		if = {
			limit = { rite_has_doctrine = doctrine_pluralism_fundamentalist }
			multiply = faith_unchanged_doctrine_cost_mult
		}
		else_if = {
			limit = { rite_has_doctrine = doctrine_pluralism_pluralistic }
			multiply = faith_changed_doctrine_cost_mult_two_step
		}
	}

	can_pick = {
		custom_description = {
			text = incompatible_tenet_pentarchy_trigger
			NOT = { tenet:tenet_pentarchy = { is_in_list = selected_doctrines } }
		}
		custom_description = {
			text = incompatible_tenet_alexandrian_catechism_trigger
			NOT = { tenet:tenet_alexandrian_catechism = { is_in_list = selected_doctrines } }
		}
		custom_description = {
			text = incompatible_tenet_adaptive_trigger
			NOT = { tenet:tenet_adaptive = { is_in_list = selected_doctrines } }
		}
	}

	parameters = {
		pluralism_fundamentalist_proselytization_bonus
		pluralism_fundamentalist_faction_join_chance
		pluralism_fundamentalist_vulnerable_to_heresy
		pluralism_righteous_revocation_tyranny_reduced
	}

	special_parameters = {
		hostility_towards_us_mult = 2
		heresy_threshold = -5
	}

	character_modifier = {
		minority_different_faith_opinion = -20
	}
}
